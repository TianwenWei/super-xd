def _sentence(text):
    text = re.sub(r'\s+', ' ', str(text or '')).strip()
    return text if not text or text[-1] in '。！？.!?' else text + '。'

def _soften_success(text, particle):
    """Change spoken presentation, preserving the complete verified fact."""
    text = text.rstrip('。.!！')
    text = text.replace('现在是', '调到', 1).replace('车窗开度', '车窗开到', 1)
    text = text.replace('已开启', '开', 1).replace('已关闭', '关', 1)
    if particle == '了':
        text = text.replace('调到', '调成', 1).replace('打开一半', '开到一半', 1)
    elif text.endswith('了'):
        text = text[:-1] + particle
    if not text.endswith(('了', '啦', '着')):
        text += particle
    return text + '。'

def render(persona, kind, detail=''):
    style = persona.id if persona.id in PERSONAS else ('professional' if persona.catchphrase_frequency == 'never' else 'playful' if persona.catchphrase_frequency == 'normal' else 'warm')
    detail = re.sub(r'\s+', ' ', str(detail or '')).strip()
    if kind == 'success':
        return _sentence(detail) if style == 'professional' else _soften_success(detail, '啦' if style == 'playful' else '了')
    if kind == 'delegated':
        goal = detail.rstrip('。.!！')
        start = {'professional': '我来', 'warm': '我帮你', 'playful': '我去'}[style]
        return f'{start}{goal}，有结果告诉你。'
    if kind == 'clarify':
        # One key question is a generation contract, never a presentation
        # cutoff: later questions can themselves carry necessary conditions.
        detail = detail.replace('可以在对应的任务卡里补充。', '').strip()
        if style != 'professional':
            particle = '呀' if style == 'playful' else '呢'
            detail = re.sub(r'([^。！？.!?])([？?])',
                            lambda match: match[1] + ('' if match[1] in '吗呢呀吧' else particle) + match[2],
                            detail, count=1)
        return _sentence(detail)
    if kind in {'refused', 'error'}:
        start = ({'professional': '未执行，', 'warm': '这次没能执行，', 'playful': '这次没做成，'}
                 if kind == 'refused' else
                 {'professional': '这次未完成，', 'warm': '这次没能完成，', 'playful': '这次没完成，'})[style]
        return start + _sentence(detail)
    if kind == 'cancel':
        return {'professional': '正在取消这项任务。', 'warm': '我正在取消这件事。', 'playful': '正在停下这件事。'}[style]
    if kind == 'updated':
        return {'professional': '补充条件已记下。', 'warm': '新条件记下了。', 'playful': '新要求记下啦。'}[style]
    # Neutral result/progress facts remain intact. Repeated result announcements
    # need no catchphrase; chat and verified control endings carry the persona.
    return _sentence(detail)

def spoken_reason(reason):
    """Keep physical refusal conditions; move protocol/schema detail to trace."""
    text = str(reason)
    if '目的地与用户原文不一致' in text:
        match = re.search(r'请确认完整地点“([^”]+)”', text)
        return f'目的地还没确认，暂时没有修改。你是要去{match.group(1)}吗？' if match else '目的地还没确认，暂时没有修改。你要去哪里？'
    if text.startswith('参数校验失败'):
        return '设置参数没确认清楚，这次没有调整。请把要求说具体一点。'
    if any(marker in text for marker in ('load_skill', 'skill_id', '未知工具', '未知函数')):
        return '这项功能暂时没能执行，请换个说法再试。'
    return text

def validation_feedback(exc, name):
    if isinstance(exc, ValidationError):
        field_name = next(iter(exc.path), None)
        spec = TOOLS[name].parameters.get('properties', {}).get(field_name, {}) if name in TOOLS else {}
        if exc.validator in {'minimum', 'maximum'} and 'minimum' in spec and 'maximum' in spec:
            label = {'volume': '音量', 'temperature': '温度', 'level': '档位', 'position': '车窗开度'}.get(field_name, '这个数值')
            unit = '度' if field_name == 'temperature' else ''
            return f'{label}只能调到{spoken_number(spec["minimum"])}到{spoken_number(spec["maximum"])}{unit}，这次没有调整。'
        return '这次的执行指令格式出了问题，没有调整。可以再试一次。'
    return spoken_reason(f'参数校验失败：{str(exc).splitlines()[0][:180]}')

    def handle(self, e):
        kind = e['type']
        if kind == '_attach':
            self.clients.add(e['outgoing'])
            e['outgoing'].put_nowait({'type': 'session.snapshot', **self.snapshot()})
            for command in self.pending_commands.values():
                e['outgoing'].put_nowait({'type': 'vehicle.command', **command['wire']})
            e['future'].set_result(True)
            self.schedule_announce()
        elif kind == '_detach':
            self.clients.discard(e['outgoing'])
            if not self.clients:
                self.asr_capture_id = None
                self.speaking = False
                if self.current: self.cancel_audio(self.current, 'control_disconnected')
        elif kind == '_audio_attach':
            self.audio_clients.add(e['outgoing'])
            e['future'].set_result(True)
        elif kind == '_audio_detach':
            self.audio_clients.discard(e['outgoing'])
            if not self.audio_clients:
                if self.current: self.cancel_audio(self.current, 'audio_disconnected')
                self.speaking = False
                self.schedule_announce()
        elif kind == 'input':
            text = str(e.get('text', '')).strip()
            if not text or len(text) > 4000:
                raise ValueError('输入需为1–4000字')
            self.asr_capture_id = None
            self.start_turn(text, str(e.get('client_request_id', ''))[:80] or None)
        elif kind == 'asr.capture':
            capture_id = e.get('capture_id')
            if not isinstance(capture_id, str) or not re.fullmatch(r'[A-Za-z0-9_-]{1,80}', capture_id):
                raise ValueError('无效语音录音标识')
            if e.get('active') is True:
                if self.asr_capture_id == capture_id:
                    return
                self.asr_capture_id = capture_id
                if self.current:
                    self.cancel_audio(self.current, 'asr_capture')
                self.speaking = False
                async def expire_capture():
                    await asyncio.sleep(90)
                    await self.post({'type': 'asr.capture', 'capture_id': capture_id, 'active': False})
                self.spawn(expire_capture())
            elif self.asr_capture_id == capture_id:
                self.asr_capture_id = None
                self.schedule_announce()
        elif kind == 'persona.set':
            if e.get('preset'):
                if e['preset'] not in PERSONAS:
                    raise ValueError('未知人设')
                self.persona = PERSONAS[e['preset']].model_copy(deep=True)
            else:
                self.persona = PersonaConfig.model_validate(e.get('config'))
            self.persona_version += 1
            self.emit('persona.changed', persona=self.persona.model_dump(), persona_version=self.persona_version)
        elif kind == 'audio.cancel':
            target = next((turn for turn in reversed(self.turns) if turn.id == e.get('turn_id')), None)
            if target and (not e.get('playback_id') or e['playback_id'] == target.playback_id):
                self.cancel_audio(target, 'client_cancelled')
                if self.current is target:
                    self.speaking = False
                    self.schedule_announce()
        elif kind == 'audio.configure':
            self.audio_enabled = e.get('enabled') is True
            if self.audio_enabled and self.tts and hasattr(self.tts, 'prewarm_stream'):
                async def warm():
                    try:
                        status = await self.tts.prewarm_stream()
                    except Exception as exc:
                        status = {'ok': False, 'error': sanitized_error(exc)}
                    await self.post({'type': '_audio_connection', 'status': status})
                self.spawn(warm())
            elif not self.audio_enabled:
                if self.current: self.cancel_audio(self.current, 'audio_disabled')
                self.speaking = False
                self.schedule_announce()
        elif kind == '_audio_connection':
            self.emit('audio.connection', **e['status'])
        elif kind == 'audio.client_metrics':
            target = next((t for t in reversed(self.turns) if t.id == e.get('turn_id') and
                           t.playback_id == e.get('playback_id')), None)
            if target and isinstance(e.get('metrics'), dict):
                metrics = {str(k)[:80]: v for k, v in list(e['metrics'].items())[:40]
                           if isinstance(v, (int, float, str, bool)) or v is None}
                target.metrics['audio_client'] = metrics
                self.trace('audio.client_metrics', {'playback_id': target.playback_id,
                           'client_request_id': target.client_request_id, 'metrics': metrics}, target,
                           status='completed' if self.audio_valid(target) else 'stale')
        elif kind == 'playback':
            if self.current and self.current.playback_id and (e.get('turn_id') != self.current.id or
                                                            e.get('playback_id') != self.current.playback_id):
                return
            if e.get('speaking') and self.current and self.current.audio_cancelled.is_set():
                return
            self.speaking = bool(e.get('speaking'))
            if not self.speaking:
                self.schedule_announce()
        elif kind == '_announce':
            self.announce()
        elif kind == '_delta':
            turn = e['turn']
            if self.is_current(turn):
                turn.metrics.setdefault('server_first_model_text_ms', (time.perf_counter() - turn.started) * 1000)
                # SSE content can precede tool_calls in the same response. It
                # is displayed immediately, but speech eligibility waits for
                # the completed routing decision to avoid premature success.
                turn.model_text_pending += e['text']
                if not (turn.task_read_required or turn.word_shape_only):
                    self.append_text(turn, e['text'], speak=False)
        elif kind == '_audio_item':
            turn, item = e['turn'], e['item']
            if not self.audio_valid(turn): return
            audio = item.get('audio')
            if isinstance(audio, bytes) and audio:
                if len(audio) % 2:
                    self.handle({'type': '_audio_error', 'turn': turn, 'error': '无效PCM音频帧'})
                    return
                elapsed = (time.perf_counter() - turn.started) * 1000
                if not turn.audio_chunks: turn.metrics['server_first_audio_ms'] = elapsed
                if any(audio) and 'server_first_nonzero_audio_ms' not in turn.metrics:
                    turn.metrics['server_first_nonzero_audio_ms'] = elapsed
                turn.audio_chunks += 1
                turn.audio_bytes += len(audio)
                self.emit_audio(turn.playback_id.encode('ascii') + audio)
                if not self.audio_clients:
                    self.cancel_audio(turn, 'audio_backpressure')
                    self.speaking = False
                    self.schedule_announce()
            else:
                diagnostics = {k: v for k, v in item.items() if k != 'audio'}
                self.emit_audio({'type': 'audio.metrics', 'turn_id': turn.id,
                                 'playback_id': turn.playback_id, 'client_request_id': turn.client_request_id,
                                 'metrics': diagnostics})
        elif kind == '_audio_finished':
            turn = e['turn']
            if not self.audio_valid(turn): return
            if not turn.audio_chunks:
                self.handle({'type': '_audio_error', 'turn': turn, 'error': 'Speech 未返回有效音频帧'})
                return
            metrics = {k: v for k, v in turn.metrics.items() if k.startswith('server_')}
            metrics.update({'audio_chunks': turn.audio_chunks, 'audio_bytes': turn.audio_bytes,
                            'speech_elapsed_ms': (time.perf_counter() - turn.started) * 1000})
            self.trace('audio.completed', {'playback_id': turn.playback_id, 'metrics': metrics}, turn,
                       status='completed')
            self.emit_audio({'type': 'audio.metrics', 'turn_id': turn.id,
                             'playback_id': turn.playback_id, 'metrics': metrics})
            self.emit_audio({'type': 'audio.ended', 'turn_id': turn.id, 'playback_id': turn.playback_id})
        elif kind == '_audio_error':
            turn = e['turn']
            if not self.audio_valid(turn): return
            self.trace('audio.error', {'playback_id': turn.playback_id, 'error': e['error']}, turn, status='failed')
            self.emit_audio({'type': 'audio.error', 'turn_id': turn.id, 'playback_id': turn.playback_id,
                             'message': e['error']})
            turn.audio_cancelled.set()
            if turn.audio_task and not turn.audio_task.done(): turn.audio_task.cancel()
            self.speaking = False
            self.schedule_announce()
        elif kind == '_model_done':
            turn = e['turn']
            if self.is_current(turn):
                turn.metrics['server_model_completed_ms'] = (time.perf_counter() - turn.started) * 1000
            self.trace('model.completed', {'model_call_index': e.get('model_call_index'),
                       'message': e['message'], 'meta': e.get('meta', {}),
                       'observed_at': e.get('observed_at', utcnow())}, turn,
                       call_id=e.get('call_id'), status='completed' if self.is_current(turn) else 'stale')
            self.model_done(e['turn'], e['message'], e.get('call_id'))
        elif kind == '_model_error':
            turn = e['turn']
            self.trace('model.error', {'model_call_index': e.get('model_call_index'), 'error': e['error'],
                       'partial': e.get('partial'), 'observed_at': e.get('observed_at', utcnow())},
                       turn, call_id=e.get('call_id'), status='failed' if self.is_current(turn) else 'stale')
            if self.is_current(turn):
                verified = self.receipt_feedback(turn, mutations_only=True)
                if verified:
                    turn.output = ''
                    self.append_text(turn, verified, source='verified_partial_receipts')
                self.append_text(turn, render(turn.persona, 'error', '后续处理连接出了问题，还有请求未确认。请稍后再试。' if verified else '这次连接出了问题，暂时没能回答。请稍后再试。'),
                                 source='error_template', original_text=e['error'])
                self.finish(turn, error=e['error'])
        elif kind == '_tool_result':
            self.tool_result(e['turn'], e['call'], e['result'])
        elif kind == '_lookup_result':
            self.accept_lookup(e['turn'], e['call'], e['result'])
        elif kind == '_vehicle_plan_done':
            self.accept_vehicle_plan(e['turn'], e['call'], e['plan'])
        elif kind == '_vehicle_plan_error':
            turn, call = e['turn'], e['call']
            self.trace('vehicle.plan.failed', {'agent': 'vehicle_planner', 'source': 'vehicle_planner',
                       'parent_call_id': call['id'], 'error': e['error']}, turn,
                       call_id=call['id'], status='failed' if self.is_current(turn) else 'stale')
            self.tool_result(turn, call, {'ok': False, 'executed': False, 'source': 'vehicle_plan',
                'code': 'vehicle_planning_failed', 'reason': e['error'],
                'spoken_reason': '这次车控计划未能核实，没有下发其中的动作。请稍后重试。'})
        elif kind == 'vehicle.execute':
            self.execute_command(e.get('command_id', ''))
        elif kind == '_command_timeout':
            self.command_timeout(e['command_id'])
        elif kind == '_task_event':
            self.task_event(e['event'])
        elif kind.startswith('task.'):
            self.explicit_task(kind[5:], e)
        else:
            raise ValueError(f'未知消息类型：{kind}')

    def emit_audio(self, message):
        # Audio never shares the control/vehicle queue. Slow consumers are
        # disconnected instead of growing an unbounded PCM backlog.
        for client in list(self.audio_clients):
            if client.qsize() >= 256:
                self.audio_clients.discard(client)
                while not client.empty(): client.get_nowait()
                client.put_nowait({'type': 'audio.error', 'message': '音频接收过慢，请重新连接'})
                client.put_nowait(None)
            else:
                client.put_nowait(message)

    def start_turn(self, text, client_request_id=None):
        if self.current:
            old = self.current
            self.cancel_audio(old, 'new_input')
            if old.model_task and not old.model_task.done():
                old.model_task.cancel()
            for planning in old.vehicle_plan_tasks.values():
                if not planning.done():
                    planning.cancel()
            for lookup in old.lookup_tasks.values():
                if not lookup.done():
                    lookup.cancel()
            for plan in old.vehicle_plans.values():
                # Already-dispatched commands retain their real late receipts;
                # no remaining action may be dispatched for an interrupted turn.
                plan['interrupted'] = True
            if not old.finished:
                old.finished = True
                old.interrupted = True
                self.trace('turn.interrupted', {'output': old.output, 'reason': 'new_input'}, old, status='interrupted')
                self.emit('stream.interrupted', turn_id=old.id)
            # Interrupted partial text belongs to that turn; pending tool records stay intact.
            if old.output and not old.group and not any(m.get('role') == 'assistant' for m in old.messages):
                old.messages.append({'role': 'assistant', 'content': old.output + ' [用户打断]'})
        self.speaking = False
        turn = Turn(uuid4().hex[:12], text, self.persona.model_copy(deep=True), self.persona_version)
        turn.loaded = list(self.loaded_tools)
        turn.setting_clauses = explicit_setting_clauses(text)
        turn.task_read_required = explicit_existing_result_request(text) and bool(self.db.list_tasks(self.id))
        turn.word_shape_only = explicit_word_shape_request(text)
        turn.client_request_id = client_request_id
        turn.messages.append({'role': 'user', 'content': text})
        self.turns.append(turn)
        self.current, self.generating = turn, True
        self.trace('turn.started', {'input': text, 'persona_version': turn.persona_version}, turn, status='running')
        self.emit('turn.started', turn_id=turn.id, text=text, persona_version=turn.persona_version,
                  client_request_id=client_request_id)
        self.start_audio(turn)
        self.launch_model(turn)

    def launch_model(self, turn):
        if turn.model_calls >= 5:
            verified = self.receipt_feedback(turn, mutations_only=True)
            if verified:
                self.append_text(turn, verified, source='verified_partial_receipts')
            self.append_text(turn, '这次没能核实所指任务的已有结果，暂时不能给出结论。'
                             if turn.task_read_required and not self.task_read_evidence(turn)
                             else '这次没能给出符合所问范围的字形回答，请再试一次。' if turn.word_scope_rejections
                             else '这次没能确认可用的功能，请把需求再具体一点。')
            self.finish(turn, error='model_step_limit')
            return
        turn.model_calls += 1
        messages, tools = self.messages_for(turn), self.tools_for(turn)
        model_call_index, call_id = turn.model_calls, uuid4().hex
        self.trace('model.started', {'model_call_index': model_call_index, 'messages': messages,
                                    'tools': tools}, turn, call_id=call_id, status='running')
        async def generate():
            partial = ''
            try:
                async for item in self.ark.stream(messages, tools):
                    if item['type'] == 'delta':
                        partial += item['text']
                        await self.post({'type': '_delta', 'turn': turn, 'text': item['text']})
                    elif item['type'] == 'message':
                        await self.post({'type': '_model_done', 'turn': turn, 'message': item['message'],
                                         'call_id': call_id, 'model_call_index': model_call_index,
                                         'meta': item.get('meta', {}), 'observed_at': utcnow()})
            except asyncio.CancelledError as exc:
                self.trace('model.cancelled', {'model_call_index': model_call_index,
                           'partial': getattr(exc, 'trace_partial', {'message': {'content': partial}, 'meta': {}})},
                           turn, call_id=call_id, status='cancelled')
            except Exception as exc:
                error = sanitized_error(exc)
                await self.post({'type': '_model_error', 'turn': turn, 'error': error,
                                'partial': getattr(exc, 'trace_partial', {'message': {'content': partial}, 'meta': {}}),
                                'call_id': call_id, 'model_call_index': model_call_index, 'observed_at': utcnow()})
        turn.model_task = self.spawn(generate())

    def model_done(self, turn, message, model_call_id=None):
        if not self.is_current(turn):
            return
        calls = message.get('tool_calls', [])
        rejection = None
        if not calls and turn.task_read_required and not self.task_read_evidence(turn):
            turn.task_read_rejections += 1
            rejection = {'code': 'task_read_required',
                         'rejected_direct_answers': turn.task_read_rejections,
                         'reason': '本轮明确要求用已有结果回答，但没有本轮真实成功manage_task(get)证据。'}
        elif not calls and turn.word_shape_only and unasked_stroke_expansion(turn.model_text_pending or message.get('content') or ''):
            turn.word_scope_rejections += 1
            rejection = {'code': 'unasked_stroke_expansion',
                         'rejected_direct_answers': turn.word_scope_rejections,
                         'reason': '用户只问带引号字词的字形，返回内容擅自展开了未问的书写步骤。'}
        if rejection:
            # The full returned message remains in model.completed. It is not
            # promoted to conversation history, UI deltas or TTS without proof.
            self.trace('model.output_rejected', {**rejection, 'model_call_id': model_call_id},
                       turn, call_id=model_call_id, status='rejected')
            turn.model_text_pending, turn.output = '', ''
            self.launch_model(turn)
            return
        turn.messages.append(message)
        if not calls:
            original = turn.model_text_pending or message.get('content') or ''
            lookup_names = {item['function']['name'] for record in turn.messages
                            for item in record.get('tool_calls', [])}
            if lookup_names == {'search_places'} and len(turn.lookup_results) == 1:
                places_result = turn.lookup_results[0]['result']
                places = places_result.get('places', [])
                if places_result.get('ok') and places:
                    # Ordinals must mean the same thing in speech, the result
                    # card and recent_places. The raw model text stays in trace.
                    ordered = '、'.join(f'{i}，{p["name"]}' for i, p in enumerate(places, 1))
                    text = f'找到{len(places)}个地点，按结果卡顺序是：{ordered}。可以告诉我名称或序号。'
                    turn.output = turn.model_text_pending = ''
                    self.append_text(turn, text, source='verified_place_results')
                    turn.metrics['output_source'] = 'verified_place_results'
                    message['content'] = text
                    self.finish(turn)
                    return
            handled = self.handled_vehicle_settings(turn)
            incomplete = not turn.vehicle_plans and 0 < handled < len(turn.setting_clauses)
            verified = self.receipt_feedback(turn, mutations_only=True) if turn.has_reads or handled or turn.vehicle_plans else ''
            if verified:
                turn.output = ''
                self.append_text(turn, verified, source='verified_partial_receipts')
                if incomplete:
                    self.append_text(turn, '还有明确设置没能确认，未完成的部分没有执行。')
                elif turn.has_reads:
                    self.append_text(turn, original, source='model_content', original_text=original)
            else:
                if turn.task_read_required or turn.word_shape_only:
                    turn.output = ''
                    self.append_text(turn, original, source='model_content', original_text=original)
                else:
                    turn.output = speech_text(original)
                    self.feed_audio(turn, original, 'model_content', original)
            turn.model_text_pending = ''
            self.finish(turn, error='incomplete_request' if incomplete else self.vehicle_plan_error(turn))
            return
        turn.model_text_pending = ''
        turn.output = ''  # Tool-call prefaces are retained in raw model trace, not final feedback.
        turn.group, turn.group_results, turn.needs_model, turn.feedback = calls, {}, False, []
        turn.route = 'tool'
        names = [call.get('function', {}).get('name') for call in calls]
        delegation_count = names.count('delegate_vehicle_control')
        group_error = None
        if delegation_count and (delegation_count > 1 or any(name in TOOLS or name == 'start_poi_navigation' for name in names)):
            group_error = {'code': 'planned_vehicle_group_conflict',
                           'reason': '整句车控只能由一层执行；委托车控不能与任何直接车控或另一委托混在同组。请只委托一次完整原句。'}
        elif any(name in TOOLS and name not in HOT_TOOL_NAMES or name == 'load_skill' for name in names):
            group_error = {'code': 'cold_vehicle_requires_delegation',
                           'reason': '非热车控只由专门车控层处理；本组没有执行。请整句调用delegate_vehicle_control，不先执行热动作、不调用load_skill。'}
        elif 'start_poi_navigation' in names and ('set_navigation_destination' in names or names.count('start_poi_navigation') > 1):
            group_error = {'code': 'navigation_group_conflict', 'reason': '一次只设置一个导航目的地，不能混用POI导航和文字目的地。'}
        if group_error:
            # Validate the routing choice for the entire group before even its
            # first hot command can be dispatched. No cross-layer replay.
            turn.needs_model = True
            for call in calls:
                turn.tool_model_calls[call['id']] = model_call_id
                self.trace('tool.started', {'name': call.get('function', {}).get('name'),
                           'arguments': call.get('function', {}).get('arguments'), 'model_call_id': model_call_id},
                           turn, call_id=call['id'], status='running')
                self.tool_result(turn, call, {'ok': False, 'executed': False,
                    'retry_model': True, **group_error})
            return
        for call in calls:
            if call.get('function', {}).get('name') in TOOLS:
                self.emit('vehicle.function_call', turn_id=turn.id, call=call)
            turn.tool_model_calls[call['id']] = model_call_id
            self.trace('tool.started', {'name': call.get('function', {}).get('name'),
                       'arguments': call.get('function', {}).get('arguments'), 'model_call_id': model_call_id},
                       turn, call_id=call['id'], status='running')
            try:
                name = call['function']['name']
                arguments = json.loads(call['function'].get('arguments', '{}'))
                if name in TOOLS:
                    if name not in HOT_TOOL_NAMES:
                        raise ValueError('非热车控必须delegate_vehicle_control')
                    validate(arguments, TOOLS[name].parameters)
                    guard = (validate_required_origin(turn.text, name, arguments, [t.text for t in self.turns[-2:-1]])
                             or validate_literal_feature_target(turn.text, name, arguments, self.vehicle)
                             or validate_literal_window_hold(turn.text, name, arguments, self.vehicle)
                             or validate_literal_authorization(turn.text, name, arguments, self.vehicle))
                    if guard:
                        self.literal_refusal(turn, call, guard)
                        continue
                    if name == 'set_navigation_destination':
                        validate_destination_literal(turn.text, arguments['destination'])
                    if TOOLS[name].state_mapping.get('operation') == 'read':
                        turn.has_reads = True
                        turn.needs_model = True  # Reading must answer the returned facts, not a generic success template.
                    self.send_command(turn, call, name, arguments)
                elif name in MAPS_BY_NAME:
                    validate(arguments, MAPS_BY_NAME[name]['function']['parameters'])
                    guard = validate_required_origin(turn.text, name, arguments, self.recent_user_inputs(turn))
                    if guard:
                        self.literal_refusal(turn, call, guard)
                        continue
                    if name in READ_TOOLS:
                        turn.has_reads = turn.needs_model = True
                        turn.route = 'lookup'
                        self.launch_lookup(turn, call, name, arguments)
                    else:
                        self.start_poi_navigation(turn, call, arguments['place_id'])
                else:
                    if name not in CONTROL_BY_NAME:
                        raise ValueError('未知工具')
                    validate(arguments, CONTROL_BY_NAME[name]['function']['parameters'])
                    if name == 'delegate_vehicle_control':
                        self.launch_vehicle_plan(turn, call, arguments)
                    else:
                        guard = validate_required_origin(turn.text, name, arguments, [t.text for t in self.turns[-2:-1]])
                        if guard:
                            self.literal_refusal(turn, call, guard)
                            continue
                        result = self.task_tool(name, arguments,
                                                request_text=turn.text if turn.task_read_required else None)
                        if result.get('retry_model'):
                            turn.needs_model = True
                        if name == 'delegate_task':
                            turn.route = 'delegate'
                        self.tool_result(turn, call, result)
            except (ValueError, KeyError, ValidationError) as exc:
                self.tool_result(turn, call, {'ok': False, 'reason': f'参数校验失败：{str(exc).splitlines()[0][:180]}',
                                             'spoken_reason': validation_feedback(exc, call.get('function', {}).get('name'))})

    def launch_lookup(self, turn, call, name, arguments):
        async def lookup():
            started = time.perf_counter()
            try:
                async with asyncio.timeout(self.lookup_timeout):
                    result = await getattr(self.maps, name)(**arguments)
            except asyncio.CancelledError:
                self.trace('tool.cancelled', {'name': name, 'model_call_id': turn.tool_model_calls.get(call['id'])},
                           turn, call_id=call['id'], status='cancelled')
                return
            except MapsError as exc:
                result = {'ok': False, 'code': exc.code, 'reason': str(exc)}
            except TimeoutError:
                result = {'ok': False, 'code': 'timeout', 'reason': '查询超时，未取得结果。'}
            except Exception:
                result = {'ok': False, 'code': 'lookup_error', 'reason': '查询服务返回异常，未取得结果。'}
            result['elapsed_ms'] = round((time.perf_counter() - started) * 1000, 1)
            await self.post({'type': '_lookup_result', 'turn': turn, 'call': call, 'result': result})
        turn.lookup_tasks[call['id']] = self.spawn(lookup())

    def launch_vehicle_plan(self, turn, call, arguments):
        snapshot = copy.deepcopy(self.vehicle)

    def accept_vehicle_plan(self, turn, call, plan):
        self.trace('vehicle.plan.received', {'agent': 'vehicle_planner', 'source': 'vehicle_planner',
                   'parent_call_id': call['id'], 'plan': plan}, turn, call_id=call['id'],
                   status='completed' if self.is_current(turn) else 'stale')
        if not self.is_current(turn):
            self.tool_result(turn, call, {'ok': False, 'executed': False, 'source': 'vehicle_plan',
                'code': 'vehicle_plan_interrupted', 'status': 'interrupted',
                'reason': '用户已打断，迟到计划没有下发。'})
            return
        if (isinstance(plan, dict) and plan.get('status') in {'refused', 'failed'} and plan.get('ok') is False
                and plan.get('intents') == [] and plan.get('actions') == [] and plan.get('keep_paths') == []
                and isinstance(plan.get('reason'), str) and plan['reason'].strip()):
            # Planner-generated transport/step failures are terminal failure
            # records, not executable plans. Preserve their original cause;
            # the execution validator still handles every nonempty action set.
            reason = plan['reason']
            original_code = plan.get('code') or plan.get('error_code')
            code = original_code or (reason if re.fullmatch(r'[a-z][a-z0-9_]*', reason) else 'vehicle_planning_failed')
            self.trace('vehicle.plan.failed', {'agent': 'vehicle_planner', 'source': 'vehicle_planner',
                       'parent_call_id': call['id'], 'code': code, 'reason': reason,
                       'planner_error': plan}, turn, call_id=call['id'], status='failed')
            self.tool_result(turn, call, {'ok': False, 'executed': False, 'source': 'vehicle_plan',
                'status': 'failed', 'code': code, 'reason': reason, 'original_error_code': original_code,
                'plan_call_id': plan.get('plan_call_id'), 'model_call_id': plan.get('model_call_id'),
                'model_calls': plan.get('model_calls'), 'intents': [], 'action_receipts': [],
                'keep_paths': [], 'read_results': [],
                'spoken_reason': '车控规划没有完成，计划中的动作没有下发。'})
            return
        try:
            # This is a fresh-state, whole-plan preflight. Simulation is never
            # accepted as an actual receipt and never mutates self.vehicle.
            validated = validate_vehicle_plan(plan, turn.text, copy.deepcopy(self.vehicle),
                completed_operations=self.completed_vehicle_operations(turn), history=self.vehicle_plan_history(turn))
            candidate = copy.deepcopy(self.vehicle)
            for action in validated.get('actions', []):
                self.validate_planned_action(turn, validated, action, candidate)
                result = execute_tool(action['name'], action['arguments'], candidate)
                if not result['ok']:
                    raise ValueError(result.get('reason', '计划动作被最新车况拒绝'))
        except (ValueError, KeyError, TypeError, ValidationError) as exc:
            self.trace('vehicle.plan.rejected', {'agent': 'vehicle_planner', 'source': 'vehicle_planner',
                       'parent_call_id': call['id'], 'reason': str(exc)}, turn,
                       call_id=call['id'], status='rejected')
            self.tool_result(turn, call, {'ok': False, 'executed': False, 'source': 'vehicle_plan',
                'code': 'vehicle_plan_invalid', 'reason': str(exc),
                'spoken_reason': '车控计划没有通过整组核对，其中的动作都没有下发。'})
            return
        if validated.get('status') != 'planned':
            question = validated.get('question') or next((item.get('question') for item in validated.get('intents', []) if item.get('question')), None)
            reason = validated.get('reason') or next((item.get('reason') for item in validated.get('intents', []) if item.get('reason')), None)
            self.tool_result(turn, call, {'ok': False, 'executed': False, 'source': 'vehicle_plan',
                'status': validated.get('status', 'refused'), 'needs_clarification': validated.get('status') == 'clarification',
                'intents': validated.get('intents', []), 'action_receipts': [],
                'keep_paths': validated.get('keep_paths', []), 'read_results': [],
                'plan_call_id': validated.get('plan_call_id'),
                'reason': question or reason or '车控需求还不能可靠确认。'})
            return
        self.trace('vehicle.plan.accepted', {'agent': 'vehicle_planner', 'source': 'vehicle_planner',
                   'parent_call_id': call['id'], 'plan': validated, 'executed': False}, turn,
                   call_id=call['id'], status='accepted')
        turn.vehicle_plans[call['id']] = {'call': call, 'plan': validated, 'index': 0,
            'action_receipts': [], 'interrupted': False, 'in_flight': None,
            'keep_baseline': {path: state_path_value(self.vehicle, path) for path in validated.get('keep_paths', [])}}
        self.dispatch_next_planned_action(turn, call['id'])

    def vehicle_plan_feedback(self, turn, result, *, mutations_only=False):
        feedback = []
        for item in result.get('action_receipts', []):
            name, body = item['name'], item['result']
            if TOOLS[name].state_mapping.get('operation') == 'read' and body.get('ok'):
                continue
            if body.get('ok'):
                feedback.append(render(turn.persona, 'success', receipt_detail(name, item['arguments'], body)))
            elif not mutations_only:
                feedback.append(render(turn.persona, 'refused', body.get('spoken_reason') or spoken_reason(body.get('reason', '动作未执行'))))
        if not mutations_only:
            if any(item.get('status') == 'kept' for item in result.get('intents', [])):
                feedback.append('要求保持的设置未调整。')
            feedback.extend(item['reason'] for item in result.get('intents', [])
                            if item.get('status') == 'failed' and item.get('keep_paths') and item.get('reason'))
            if not result.get('action_receipts') and not result.get('ok'):
                feedback.append(render(turn.persona, 'clarify' if result.get('needs_clarification') else 'refused',
                    result.get('spoken_reason') or spoken_reason(result.get('reason', '车控计划未执行。'))))
        return feedback

    def task_tool(self, name, args, request_text=None):
        if name == 'delegate_task':
            # A typed product profile, not weather detection or a router.
            # Reject the original incompatible parameters before any task
            # exists; only the same Talker may choose corrected arguments.
            if args.get('query_domain') == 'weather' and args.get('query_mode') != 'quick':
                return {'ok': False, 'executed': False, 'retry_model': True,
                        'code': 'weather_quick_required',
                        'delegation_policy_guard': {'kind': 'typed_weather_profile',
                            'query_domain': 'weather', 'requested_query_mode': args.get('query_mode'),
                            'supported_query_mode': 'quick'},
                        'reason': '本原型weather入口只支持显式query_mode=quick，包括官网天气查询。当前参数不符合入口配置，尚未登记任务；请由同一Talker重新核对并选择参数，不能改走非天气领域冒充。'}
            task = self.db.create_task(self.id, args['goal'],
                                       context=self.task_context(self.current.id if self.current else None,
                                                                 args.get('read_details') is True,
                                                                 args.get('query_mode', 'research'), args.get('spoken_goal'),
                                                                 args.get('query_domain', 'general')))
            self.emit('ui.card', task=task, origin_turn_id=self.current.id if self.current else None)
            goal = speech_text(args.get('spoken_goal') or task['goal']).rstrip('。')
            return {'ok': True, 'task': task, 'feedback_kind': 'delegated', 'detail': goal}
        if 'task_ids' in args:
            if args.get('action') != 'get' or args.get('task_id') is not None:
                return {'ok': False, 'reason': 'task_ids只用于批量get，不能同时指定task_id。'}
            identities = args['task_ids']
            if (not isinstance(identities, list) or not 1 <= len(identities) <= 8
                    or any(not isinstance(identity, str) or not identity.strip() for identity in identities)
                    or len(set(identities)) != len(identities)):
                return {'ok': False, 'reason': '请指定一到八个不同任务ID。'}
            selected = []
            for task_id in args['task_ids']:
                task, error = self.resolve_task(task_id)
                if error:
                    return error
                scope_error = self.task_reference_scope_error(request_text, task)
                if scope_error:
                    return scope_error
                selected.append(task)
            items = [self.task_tool('manage_task', {'action': 'get', 'task_id': task['id'],
                                                   'read_details': args.get('read_details') is True}) for task in selected]
            detail = '\n'.join(f"{(task.get('context') or {}).get('spoken_goal') or task['goal']}：{item['detail']}" for task, item in zip(selected, items))
            return {'ok': True, 'tasks': selected, 'feedback_kind': 'progress', 'detail': detail,
                    'items': items}
        task, error = self.resolve_task(args.get('task_id'))
        if error:
            return error
        if args.get('action') == 'get':
            scope_error = self.task_reference_scope_error(request_text, task)
            if scope_error:
                return scope_error
            result = task.get('result')
            if task['status'] == 'completed' and isinstance(result, dict):
                detail = result.get('summary', '') if args.get('read_details') is True else result.get('spoken_summary') or brief_result(result.get('summary', ''))
            else:
                detail = task.get('question') or task['summary']
                if re.search(r'正在推理.*第\s*\d+\s*步', detail):
                    detail = '还在分析，结果出来就告诉你。'
            return {'ok': True, 'task': task, 'feedback_kind': 'progress', 'detail': detail}
        action = args['action']
        try:
            updated = self.db.mutate_task(task['id'], action, args.get('text', ''), args.get('expected_revision'))
            self.emit('ui.card', task=updated)
            self.announcements.pop(task['id'], None)
            return {'ok': True, 'task': updated, 'feedback_kind': 'cancel' if action == 'cancel' else 'updated',
                    'detail': '正在取消这项任务。' if action == 'cancel' else '新条件记下了，会按这些要求继续。'}
        except ValueError as exc:
            return {'ok': False, 'reason': str(exc)}

    def validate_planned_action(self, turn, plan, action, state):
        name, arguments = action['name'], action['arguments']
        if name not in TOOLS:
            raise ValueError('计划包含非车控函数')
        validate(arguments, TOOLS[name].parameters)
        guard = (validate_required_origin(turn.text, name, arguments, [t.text for t in self.turns[-2:-1]])

                 or validate_literal_window_hold(turn.text, name, arguments, state)
                 or validate_literal_authorization(turn.text, name, arguments, state))
        if guard:
            raise ValueError(guard['reason'])
        if name == 'set_navigation_destination':
            intent = next(item for item in plan['intents'] if item['id'] == action['intent_id'])
            validate_destination_literal(intent.get('source_text') or intent.get('text') or turn.text,
                                         arguments['destination'])

    def execute_planned_command(self, turn, parent_id, action):
        """Validate and commit in one synchronous actor operation at real ACK.

        A successful dispatch-time simulation is not authority to commit: old
        already-dispatched commands may receive their ACK in between.
        """
        execution = turn.vehicle_plans[parent_id]
        base = {'ok': False, 'executed': False, 'name': action['name'],
                'arguments': copy.deepcopy(action['arguments']), 'source': 'vehicle_plan'}
        try:
            outcomes = {item['action_id']: item['status'] for item in execution['action_receipts']}
            if any(outcomes.get(dependency) != 'completed' for dependency in action.get('depends_on', [])):
                return {**base, 'code': 'vehicle_plan_dependency_failed',
                        'reason': '依赖动作未取得成功回执，本动作没有执行。', 'state': copy.deepcopy(self.vehicle)}
            constraints = [{'path': path, 'expected': copy.deepcopy(expected),
                            'actual': state_path_value(self.vehicle, path)}
                           for path, expected in execution['keep_baseline'].items()]
            if any(not same_state_value(item['expected'], item['actual']) for item in constraints):
                return {**base, 'code': 'vehicle_plan_keep_changed', 'keep_constraints': constraints,
                        'reason': '要求保持的设置在等待期间发生了变化，本动作没有执行。',
                        'state': copy.deepcopy(self.vehicle)}
            try:
                validate_vehicle_action_keep(action['name'], action['arguments'], self.vehicle,
                                             execution['plan'].get('keep_paths', []))
            except ValueError as exc:
                return {**base, 'code': 'vehicle_plan_keep_conflict', 'validation_reason': str(exc),
                        'reason': '按最新车况执行会写入要求保持的设置，本动作没有执行。',
                        'state': copy.deepcopy(self.vehicle)}
            self.validate_planned_action(turn, execution['plan'], action, self.vehicle)
            candidate = copy.deepcopy(self.vehicle)
            result = execute_tool(action['name'], action['arguments'], candidate)
            if not result.get('ok'):
                return {**result, 'executed': False, 'source': 'vehicle_plan'}
            # This catches dynamic side effects omitted from a static mapping,
            # such as temperature propagation after zone-sync was enabled.
            changed = [{'path': path, 'expected': copy.deepcopy(expected),
                        'proposed': state_path_value(candidate, path)}
                       for path, expected in execution['keep_baseline'].items()
                       if not same_state_value(expected, state_path_value(candidate, path))]
            if changed:
                return {**base, 'code': 'vehicle_plan_keep_conflict', 'keep_constraints': changed,
                        'reason': '按最新车况执行会改变要求保持的设置，本动作没有执行。',
                        'state': copy.deepcopy(self.vehicle)}
            self.vehicle.clear()
            self.vehicle.update(candidate)
            return {**result, 'executed': True, 'source': 'vehicle_plan', 'state': copy.deepcopy(self.vehicle)}
        except (ValueError, KeyError, TypeError, ValidationError) as exc:
            return {**base, 'code': 'vehicle_plan_commit_rejected', 'reason': str(exc),
                    'state': copy.deepcopy(self.vehicle)}

    def dispatch_next_planned_action(self, turn, parent_id):
        execution = turn.vehicle_plans[parent_id]
        if execution['in_flight']:
            return
        actions = execution['plan']['actions']
        while execution['index'] < len(actions):
            action = actions[execution['index']]
            execution['index'] += 1
            receipt = {'action_id': action['id'], 'intent_id': action['intent_id'],
                       'name': action['name'], 'arguments': copy.deepcopy(action['arguments'])}
            if not self.is_current(turn) or execution['interrupted']:
                execution['interrupted'] = True
                execution['action_receipts'].append({**receipt, 'status': 'interrupted',
                    'result': {'ok': False, 'executed': False, 'reason': '用户已打断，剩余动作没有下发。'}})
                continue
            outcomes = {item['action_id']: item['status'] for item in execution['action_receipts']}
            if any(outcomes.get(dependency) != 'completed' for dependency in action.get('depends_on', [])):
                execution['action_receipts'].append({**receipt, 'status': 'blocked',
                    'result': {'ok': False, 'executed': False, 'reason': '依赖动作未成功，未执行本动作。'}})
                continue
            try:
                self.validate_planned_action(turn, execution['plan'], action, self.vehicle)
                probe = execute_tool(action['name'], action['arguments'], copy.deepcopy(self.vehicle))
                if not probe['ok']:
                    raise ValueError(probe.get('reason', '当前车况不允许本动作'))
            except (ValueError, KeyError, TypeError, ValidationError) as exc:
                execution['action_receipts'].append({**receipt, 'status': 'failed',
                    'result': {'ok': False, 'executed': False, 'reason': str(exc)}})
                continue
            child = {'id': f"{parent_id}:{action['id']}", 'type': 'function',
                     'function': {'name': action['name'], 'arguments': json.dumps(action['arguments'], ensure_ascii=False)}}
            execution['in_flight'] = action['id']
            turn.tool_model_calls[child['id']] = execution['plan'].get('model_call_id') or execution['plan'].get('plan_call_id')
            self.trace('vehicle.plan.action_started', {'agent': 'vehicle_planner', 'source': 'vehicle_plan',
                       'parent_call_id': parent_id, 'plan_call_id': execution['plan'].get('plan_call_id'),
                       'action': action}, turn, call_id=child['id'], status='running')
            self.emit('vehicle.function_call', turn_id=turn.id, call=child, source='vehicle_plan',
                      parent_call_id=parent_id, intent_id=action['intent_id'], action_id=action['id'])
            self.send_command(turn, child, action['name'], action['arguments'],
                              plan_parent_id=parent_id, plan_action=action)
            return
        self.finish_vehicle_plan(turn, parent_id)

    def finish_vehicle_plan(self, turn, parent_id):
        execution = turn.vehicle_plans[parent_id]
        plan, receipts = execution['plan'], execution['action_receipts']
        intents = []
        for original in plan['intents']:
            intent = copy.deepcopy(original)
            bound = [item for item in receipts if item['intent_id'] == intent['id']]
            if intent['status'] == 'completed':
                # Validator bound these IDs to actual earlier receipts from
                # this turn; the Pro cannot declare success from prose.
                pass
            elif intent['status'] == 'keep':
                unchanged = all(same_state_value(state_path_value(self.vehicle, path), execution['keep_baseline'][path])
                                for path in intent['keep_paths'])
                intent['status'] = 'kept' if unchanged else 'failed'
                if not unchanged:
                    intent['reason'] = '要求保持的设置在等待期间发生了变化，不能确认保持成功。'
            elif bound and all(item['status'] == 'completed' for item in bound):
                intent['status'] = 'completed'
            else:
                intent['status'] = next((item['status'] for item in bound if item['status'] != 'completed'), 'failed')
                intent['reason'] = next((item['result'].get('reason') for item in bound if not item['result'].get('ok')), '未取得实际执行回执。')
            intents.append(intent)
        reads = [copy.deepcopy(item) for item in receipts if TOOLS[item['name']].state_mapping.get('operation') == 'read'
                 and item['result'].get('ok')]
        ok = all(intent['status'] in {'completed', 'kept'} for intent in intents)
        result = {'ok': ok, 'executed': any(item['result'].get('ok') and
                    TOOLS[item['name']].state_mapping.get('operation') != 'read' for item in receipts),
                  'status': 'completed' if ok else 'interrupted' if execution['interrupted'] else 'partial',
                  'source': 'vehicle_plan', 'plan_call_id': plan.get('plan_call_id'),
                  'intents': intents, 'action_receipts': copy.deepcopy(receipts),
                  'keep_paths': copy.deepcopy(plan.get('keep_paths', [])), 'read_results': reads,
                  'keep_constraints': [{'path': path, 'expected': copy.deepcopy(expected),
                                        'actual': state_path_value(self.vehicle, path),
                                        'unchanged': same_state_value(expected, state_path_value(self.vehicle, path))}
                                       for path, expected in execution['keep_baseline'].items()]}
        if reads and self.is_current(turn):
            turn.has_reads = True
            turn.needs_model = True
        self.tool_result(turn, execution['call'], result)

    def task_reference_scope_error(self, text, task):
        # A bounded literal rejection of the selected ID, not a replacement
        # task lookup. General task references remain the model's decision.
        # Coordinated requests can legitimately read differently sourced
        # tasks. This limited guard only covers an unambiguous single-task
        # reference, and does not try to resolve a compound request itself.
        if (len(re.findall(r'任务', text or '')) > 1
                or re.search(r'和|以及|还有|分别|同时|都|(?:两|多|各)[个项]', text or '')):
            return None
        match = next((item for item in re.finditer(r'法国气象局官网(?:那个|那项)任务', text or '')
                      if not re.search(r'(?:不要(?:读|看|查)?|别(?:读|看|查)?|不是|并非|不必(?:读|看|查)?)\s*$',
                                       (text or '')[max(0, item.start()-12):item.start()])), None)
        goal = task['goal']
        domains = (task.get('context') or {}).get('required_source_domains') or []
        declared_official = isinstance(domains, list) and any(
            isinstance(domain, str) and (domain.casefold() == 'meteofrance.com'
                                         or domain.casefold().endswith('.meteofrance.com')) for domain in domains)
        official_goal = (('法国气象局' in goal and '官网' in goal)
                         or re.search(r'(?<![A-Za-zÀ-ÖØ-öø-ÿ])m[ée]t[ée]o[-\s]+france(?![A-Za-z])', goal, re.I)
                         or re.search(r'(?<![A-Za-z0-9_.-])(?:[A-Za-z0-9_-]+\.)*meteofrance\.com(?![A-Za-z0-9_.-])', goal, re.I))
        if not match or declared_official or official_goal:
            return None
        return {'ok': False, 'executed': False, 'retry_model': True,
                'code': 'task_reference_scope_mismatch',
                'task_reference_guard': {'kind': 'literal_source_scope', 'selected_task_id': task['id'],
                    'selected_revision': task['revision'], 'selected_goal': task['goal'],
                    'requested_scope': '法国气象局官网',
                    'evidence': [{'start': match.start(), 'end': match.end(), 'text': match.group()}]},
                'reason': '所读任务不包含用户明确指定的法国气象局官网要求，不能把另一任务结果当作该官方任务结论。请重新核对用户所指的任务ID。'}

    def resolve_task(self, task_id=None):
        tasks = self.db.list_tasks(self.id)
        if not tasks:
            return None, {'ok': False, 'code': 'no_task', 'retry_model': True, 'executed': False,
                          'reason': '本会话尚无后台任务，不能manage_task。请根据本轮原句重新选择直接回答或实际可用功能；车控缺功能时delegate_vehicle_control，不要创建无关后台任务。'}
        if task_id:
            found = next((t for t in tasks if t['id'] == task_id), None)
            return found, None if found else {'ok': False, 'reason': '此会话没有该任务'}
        active = [t for t in tasks if t['status'] not in {'completed', 'failed', 'cancelled'}]
        if len(active) == 1:
            return active[0], None
        if not active and len(tasks) == 1:
            return tasks[0], None
        return None, {'ok': False, 'needs_clarification': True, 'reason': '请明确指哪个任务',
                      'candidates': [{'id': t['id'], 'goal': t['goal'], 'status': t['status']} for t in (active or tasks)[-8:]]}

    def explicit_task(self, action, event):
        if action == 'create':

                                      event.get('expected_revision'))
        self.announcements.pop(task['id'], None)
        self.emit('ui.card', task=updated)

    def execute_command(self, command_id):
        command = self.pending_commands.pop(command_id, None)
        if not command:
            return  # duplicates and timed out commands cannot execute twice
        wire, turn = command['wire'], command['turn']
        if command.get('plan_parent_id') is not None:
            result = self.execute_planned_command(turn, command['plan_parent_id'], command['plan_action'])
            if not result.get('ok'):
                self.trace('vehicle.plan.commit_rejected', {'agent': 'vehicle_planner', 'source': 'vehicle_plan',
                           'parent_call_id': command['plan_parent_id'], 'action': command['plan_action'],
                           'command_id': command_id, 'result': result}, turn,
                           call_id=command['call']['id'], status='stale' if not self.is_current(turn) else 'rejected')
        else:
            result = execute_tool(wire['name'], wire['arguments'], self.vehicle)
        poi = turn.poi_commands.get(command['call']['id'])
        if result.get('ok') and poi:
            self.vehicle['navigation']['destination_place'] = copy.deepcopy(poi)
            result['state'] = copy.deepcopy(self.vehicle)
            result.update(feedback_kind='result', detail=f'已将模拟导航目的地设为{poi["name"]}，未计算实际路线。',
                          place=copy.deepcopy(poi))
        latency = (time.perf_counter() - turn.started) * 1000
        self.completed_commands[command_id] = result
        attribution = {key: wire[key] for key in ('source', 'parent_call_id', 'plan_call_id', 'action_id', 'intent_id') if key in wire}
        self.trace('vehicle.receipt', {**attribution, 'command_id': command_id, 'result': result, 'latency_ms': latency,
                   'late': not self.is_current(turn)}, turn, call_id=command['call']['id'],
                   status='stale' if not self.is_current(turn) else 'completed' if result['ok'] else 'failed')
        if len(self.completed_commands) > 500:
            self.completed_commands.pop(next(iter(self.completed_commands)))
        turn.commands[-1 if len(turn.commands) == 1 else next(i for i,c in enumerate(turn.commands) if c['command_id'] == command_id)].update({'receipt_ms': latency, 'ok': result['ok']})
        self.emit('vehicle.result', command_id=command_id, turn_id=turn.id, result=result, latency_ms=latency)
        self.emit('vehicle.state', state=self.vehicle)
        if command.get('plan_parent_id') is not None:
            self.planned_action_result(turn, command['plan_parent_id'], command['plan_action'], result)
        else:
            self.tool_result(turn, command['call'], result)

    def command_timeout(self, command_id):
        command = self.pending_commands.pop(command_id, None)
        if not command:
            return
        turn = command['turn']
        result = {'ok': False, 'reason': '模拟车机未在时限内确认，未执行；请检查浏览器连接。'}
        attribution = {key: command['wire'][key] for key in ('source', 'parent_call_id', 'plan_call_id', 'action_id', 'intent_id') if key in command['wire']}
        self.trace('vehicle.receipt', {**attribution, 'command_id': command_id, 'result': result, 'timeout': True,

                   status='stale' if not self.is_current(turn) else 'failed')
        self.emit('vehicle.result', command_id=command_id, turn_id=turn.id, result=result,
                  latency_ms=(time.perf_counter() - turn.started) * 1000)
        if command.get('plan_parent_id') is not None:
            self.planned_action_result(turn, command['plan_parent_id'], command['plan_action'], result)
        else:
            self.tool_result(turn, command['call'], result)

    def tool_result(self, turn, call, result):
        if call['id'] in turn.group_results:
            return
        name = call['function']['name']
        if name == 'delegate_vehicle_control' and result.get('source') == 'vehicle_plan':
            result = copy.deepcopy(result)
            result.setdefault('status', 'completed' if result.get('ok') else 'failed')
            result.setdefault('intents', [])
            result.setdefault('action_receipts', [])
            result.setdefault('read_results', [])
            result.setdefault('keep_paths', [])
            execution = turn.vehicle_plans.setdefault(call['id'], {'call': call,

    def receipt_feedback(self, turn, mutations_only=False):
        feedback = []
        all_calls = [call for message in turn.messages for call in message.get('tool_calls', [])
                     if call['id'] in turn.tool_results and
                     call['function']['name'] != 'load_skill']
        all_calls = [call for call in all_calls if
                     not turn.tool_results[call['id']].get('retry_model') and
                     not (call['function']['name'] in TOOLS and TOOLS[call['function']['name']].state_mapping.get('operation') == 'read')
                     and (not mutations_only or call['function']['name'] in TOOLS
                          or call['function']['name'] in {'delegate_vehicle_control', 'start_poi_navigation'})
                     and call['function']['name'] not in READ_TOOLS]
        for item in all_calls:
            body = turn.tool_results[item['id']]
            if item['function']['name'] == 'delegate_vehicle_control' and body.get('source') == 'vehicle_plan':
                feedback.extend(self.vehicle_plan_feedback(turn, body, mutations_only=mutations_only))
                continue
            if body.get('authorization_guard'):
                feedback.append(render(turn.persona, 'clarify' if body.get('needs_clarification') else 'result',
                                       body['spoken_reason']))
                continue
            if body.get('ok', True):
                if body.get('feedback_kind'):
                    feedback.append(render(turn.persona, body['feedback_kind'], body['detail']))
                elif item['function']['name'] in TOOLS:
                    # Values are from the verified receipt, never generated claims.
                    feedback.append(render(turn.persona, 'success', receipt_detail(item['function']['name'], json.loads(item['function']['arguments']), body)))
            else:
                detail = body.get('spoken_reason') or spoken_reason(body.get('reason', body.get('error', '执行器拒绝')))
                if body.get('candidates'):
                    detail = '你是指哪项任务？可以在对应的任务卡里补充。'
                feedback.append(render(turn.persona, 'clarify' if body.get('needs_clarification') else 'refused', detail))
        successful_vehicle_only = all_calls and all(
            item['function']['name'] in TOOLS and turn.tool_results[item['id']].get('ok')
            for item in all_calls)
        return ('，'.join(part.rstrip('。.!！') for part in feedback) + '。'
                if len(feedback) > 1 and successful_vehicle_only else '\n'.join(feedback))

def _weather_finish_guard(args, task, sources, observations):
    if not _weather_task(task):
        return
    goal = task.get("goal", "")
    domains = _required_source_domains(task)
    records = _verified_forecast_records(observations)
    urls = args.get("source_urls", list(sources)[:20])
    adopted = [record for record in records if record.get("source_url") in urls]
    bodies = (args["summary"], args.get("spoken_summary") or args["summary"])
    for body in bodies:
        normalized, claims = _temperature_claims(body)
        if claims and not adopted:
            raise ValueError("天气温度数字须有本版本成功get_weather_forecast及一致raw记录；请核实后再finish，未核实说明不能豁免数字。")
        if claims and re.search(r"°F|华氏", normalized, re.I):
            raise ValueError("天气数字须保留实际forecast的摄氏单位，不能混用华氏数值。")
        if claims and domains and not re.search(r"open[- ]?meteo", body, re.I) and not _official_limit(body):
            raise ValueError("指定官网任务的实际温度来自Open-Meteo；报告和口语须明确提供方或说明官网温度未核实，不能混淆来源。")
        tolerance = .51 if re.search(r"大约|约合|大概|约\s*[-\d]|about|approximately", normalized, re.I) else 1e-6
        for role, value in claims:
            key = "temperature_min" if role == "minimum" else "temperature_max"
            if not any(abs(value - record[key]) <= tolerance if role != "value" else
                       min(abs(value - record["temperature_min"]), abs(value - record["temperature_max"])) <= tolerance for record in adopted):
                raise ValueError("天气报告的温度数字与本版本实际forecast最低/最高值不一致，不能提交未观测数值。")
        # Explicit forecast dates must match adopted observations; query/update
        # timestamps are metadata, not claims about the forecast day.
        for clause in re.split(r"[，,。；;\n]", normalized):
            for mention in re.finditer(r"(?<!\d)\d{4}-\d{2}-\d{2}(?!\d)", clause):
                if re.search(r"(?:查询时间|读取时间|更新于|更新时间)[：:\s]*$", clause[:mention.start()]):
                    continue
                if claims and adopted and mention.group() not in {record["date"] for record in adopted}:
                    raise ValueError("天气报告日期与实际forecast日期不一致。")
        if claims and re.search(r"明天|后天|今天", task.get("goal", "")):
            captured = (task.get("context") or {}).get("captured_at")
            if captured:
                offset = 2 if "后天" in task["goal"] else 1 if "明天" in task["goal"] else 0
                if not any(record["date"] == (datetime.fromisoformat(captured.replace("Z", "+00:00")).astimezone(
                        ZoneInfo(record["timezone"])).date() + timedelta(days=offset)).isoformat() for record in adopted):
                    raise ValueError("实际forecast日期不符合本版本任务的当地目标日。")
        for clause in re.split(r"[，,。；;\n]", body):
            official_name = re.search(r"官方|官网|法国气象局|Météo[- ]?France", clause, re.I)
            # A separate sentence may explicitly attribute the preceding
            # temperatures, but merely saying a page was consulted is not
            # a temperature attribution.
            attribution = re.search(r"^(?:该)?(?:数字|数据|温度|气温|预报)?(?:来自|出自|来源(?:于|为|是)?)", clause.strip())
            if (claims and official_name and (_temperature_claims(clause)[1] or attribution)
                    and not _official_limit(clause) and not re.search(r"并非|不是|非官方|不属于", clause)):
                raise ValueError("实际Open-Meteo温度不能标为气象局官方数值；官网读取不等于核实该温度。")
            if re.search(r"open[- ]?meteo", clause, re.I) and re.search(r"官方|法国气象局|Météo[- ]?France", clause, re.I):
                if not _official_limit(clause) and not re.search(r"并非|不是|非官方|不属于", clause):
                    raise ValueError("Open-Meteo不是法国气象局官网预报，不能标为官方来源。")
    for domain in set(domains):
        official = [source for source in sources.values() if source.get("extracted") is True and
                    ((urlparse(source.get("url", "")).hostname or "") == domain or
                     (urlparse(source.get("url", "")).hostname or "").endswith("." + domain))]
        if not official and not all(_official_limit(body) for body in bodies):
            raise ValueError("未真实读取任务要求的官方域名页面；请在预算内核实，或在完整报告和口语中明确官方来源未完整核实，不得称已查询确认。")

    def _boundary(self, task_id: str, revision: int | None = None) -> dict[str, Any]:
        task = self.db.apply_pending(task_id)
        if not task:
            raise RuntimeError("Task no longer exists")
        if task["status"] in ("cancelling", "cancelled"):
            self.db.save_task(task_id, task["revision"], status="cancelled", summary="任务已取消。", question=None)
            raise TaskCancelled()
        if task["status"] in ("completed", "failed"):
            # A concurrent deadline sweep may have closed this task while an
            # external request was in flight. Never resurrect it.
            raise TaskCancelled()
        if revision is not None and task["revision"] != revision:
            raise RevisionChanged(task)
        return task

    async def run_task(self, task: dict[str, Any]) -> None:
        started = time.perf_counter()
        mode = self._query_mode(task)
        deadline_reason = "absolute_deadline"
        try:
            # A reply/new revision must never reset the absolute task deadline.
            remaining = self.timeout_seconds
            if task.get("created_at"):
                created = datetime.fromisoformat(task["created_at"].replace("Z", "+00:00"))
                if created.tzinfo is None:
                    created = created.replace(tzinfo=timezone.utc)
                remaining -= max(0, (datetime.now(timezone.utc) - created).total_seconds())
            if remaining <= 0:
                raise asyncio.TimeoutError()
            if mode == "quick":
                spent, _ = self._quick_usage(task)
                execution_remaining = self.quick_timeout_seconds - spent
                if execution_remaining < remaining:
                    remaining, deadline_reason = execution_remaining, "quick_execution_budget"
                if remaining <= 0:
                    raise asyncio.TimeoutError()
            await asyncio.wait_for(self._run_task(task), remaining)
        except TaskCancelled:
            return
        except asyncio.TimeoutError:
            current = self.db.get_task(task["id"]) or task
            self._trace(current, "task.timeout", {"reason": deadline_reason, "query_mode": mode},
                        step=current.get("steps"), status="failed")
            error = (f"快速查询累计执行超过{self.quick_timeout_seconds:g}秒时间上限，已停止；没有宣称完成。"
                     if deadline_reason == "quick_execution_budget" else "任务超过时间上限，已停止；没有宣称完成。")
            self._fail_current(task["id"], error)
        except asyncio.CancelledError:
            # Process shutdown is outside the promised continuous-online scope.
            self._trace(task, "worker.interrupted", {"reason": "worker_coroutine_cancelled"},
                        step=task.get("steps"), status="cancelled")
            raise
        except Exception as exc:
            # Unexpected setup errors still belong to the claimed revision. An
            # old failure must never terminate subsequently updated conditions.
            self._trace(task, "step.error", {"error": _safe_error(exc, None), "phase": "setup"},
                        step=task.get("steps"), status="error")
            self._fail_revision(task["id"], task["revision"], _safe_error(exc))
        finally:
            if mode == "quick":
                current = self.db.get_task(task["id"]) or task
                self._trace(current, "task.execution.finished", {"query_mode": mode,
                    "elapsed_seconds": time.perf_counter() - started},
                    step=current.get("steps"), status=current["status"])

    async def _quick_lookup(self, args, sources, suggestions, observations, task, revision,
                            step, model_call_id, parent_call_id):
        async def invoke(name, arguments):
            self._boundary(task["id"], revision)
            call_id = uuid4().hex
            self._trace(task, "tool.started", {"name": name, "arguments": arguments,
                "parent_call_id": parent_call_id}, revision=revision, step=step, call_id=call_id,
                model_call_id=model_call_id, status="running")
            try:
                result = await self._execute_tool(name, arguments, sources, suggestions, observations,
                                                  source_domains=_required_source_domains(task))
            except (RevisionChanged, TaskCancelled):
                raise
            except BaseException as exc:
                status = "cancelled" if isinstance(exc, asyncio.CancelledError) else self._response_status(task["id"], revision, "error")
                self._trace(task, "tool.error", {"name": name, "arguments": arguments,
                    "error": _safe_error(exc, None), "parent_call_id": parent_call_id},
                    revision=revision, step=step, call_id=call_id, model_call_id=model_call_id, status=status)
                observations.append({"tool": name, "ok": False, "attempted": True, "error": _safe_error(exc)})
                raise
            self._trace(task, "tool.completed", {"name": name, "arguments": arguments,
                "result": result, "parent_call_id": parent_call_id}, revision=revision, step=step,
                call_id=call_id, model_call_id=model_call_id,
                status=self._response_status(task["id"], revision, "completed"))
            self._boundary(task["id"], revision)
            return result

        search = await invoke("search_web", {"query": args["query"], "max_results": args.get("max_results", 3)})
        urls = []
        domains = _required_source_domains(task)
        for row in search.get("results", []):
            url = row.get("url") if isinstance(row, dict) else None
            if domains and (not isinstance(url, str) or not _source_in_domains(url, domains)):
                continue
            if url in sources and url not in urls:
                urls.append(url)
            if len(urls) == 2:
                break
        if not urls:
            return {"ok": True, "search": search, "extract": None,
                    "notice": "真实搜索没有获得可读取来源，无法确认实时事实。"}
        try:
            extract = await invoke("extract_web", {"urls": urls})
        except (RevisionChanged, TaskCancelled):
            raise
        except Exception as exc:
            extract = {"ok": False, "error": _safe_error(exc)}
        return {"ok": True, "search": search, "extract": extract}

    async def _execute_tool(self, name: str, args: dict[str, Any], sources: dict[str, Any],
                            suggestions: list[dict[str, Any]], observations: list[dict[str, Any]], *,
                            source_domains=None) -> dict[str, Any]:
        if name == "get_weather_forecast":
            if self.weather is None:
                from .weather import WeatherClient
                self.weather = WeatherClient()
            data = await self.weather.forecast(args["city"], args["date"], country_code=args.get("country_code"))
            if data.get("ok") is False:
                observations.append({"tool": name, "ok": False, "attempted": True,
                                     "needs_clarification": bool(data.get("needs_clarification"))})
                return data
            if data.get("ok") is not True or data.get("data_type") != "forecast" or data.get("date") != args["date"]:
                raise ValueError("天气工具未返回所请求日期的真实forecast记录，不能当预报。")
            low, high = data.get("temperature_min"), data.get("temperature_max")
            if any(type(value) not in (int, float) or not math.isfinite(value) for value in (low, high)) or low > high:
                raise ValueError("天气工具的温度记录无效，不能当预报。")
            raw_forecast = (data.get("raw") or {}).get("forecast")
            if not isinstance(raw_forecast, dict) or not _forecast_raw_matches(data, raw_forecast):
                raise ValueError("天气工具的date/timezone/摄氏单位或温度与原始raw不一致，不能补造或当作已验证预报。")

            if parsed.scheme not in ("http", "https") or not parsed.netloc or not data.get("queried_at") or not data.get("timezone") or data.get("unit") != "°C":
                raise ValueError("天气工具缺少真实来源、查询时间、时区或摄氏单位，不能当预报。")
            geo_url = data.get("geocoding_source_url")
            if geo_url:
                geo = urlparse(geo_url)
                if geo.scheme not in ("http", "https") or not geo.netloc:
                    raise ValueError("天气工具的地理编码来源 URL 无效。")
            record = {key: deepcopy(data[key]) for key in ("date", "location", "timezone", "temperature_min", "temperature_max", "unit", "source_url", "queried_at")}
            sources[url] = {"url": url, "title": "Open-Meteo结构化天气预报", "queried_at": data["queried_at"],
                            "kind": "forecast", "verified": True, "extracted": False,
                            "forecast_date": data["date"], "timezone": data["timezone"], "unit": data["unit"]}
            if geo_url:
                sources[geo_url] = {"url": geo_url, "title": "Open-Meteo城市地理编码", "queried_at": data["queried_at"],
                                    "kind": "geocoding", "verified": True, "extracted": False,
                                    "supports": "location"}
            observations.append({"tool": name, "ok": True, "attempted": True,
                                 "record": record, "queried_at": data["queried_at"],
                                 "forecast_verified": True, "raw_forecast": deepcopy(raw_forecast)})
            return data
        if name == "search_web":
            options = {'include_domains': source_domains} if source_domains else {}
            data = await self.tavily.search(args["query"], max_results=args.get("max_results", 5), **options)
            self._collect_sources(data, sources, "search")
            observations.append({"tool": name, "query": args["query"], "ok": True, "attempted": True,
                                 "result_count": len(data.get("results", [])), "queried_at": data.get("queried_at")})
            return {"ok": True, **data}
        if name == "extract_web":
            for url in args["urls"]:
                parsed = urlparse(url)
                if parsed.scheme not in ("https", "http") or not parsed.netloc or url not in sources:
                    raise ValueError("只能读取本任务真实搜索获得的 http(s) 来源 URL。")
            data = await self.tavily.extract(args["urls"])
            self._collect_sources(data, sources, "extract")
            observations.append({"tool": name, "ok": True, "attempted": True, "result_count": len(data.get("results", [])),
                                 "failed_results": data.get("failed_results", []), "queried_at": data.get("queried_at")})
            return {"ok": True, **data}
        if name == "load_skill":
            from .catalog import load_skill
            return load_skill(args["skill_id"])
        if name == "ask_clarification":
            return {"ok": True, "question": args["question"]}
        if name == "suggest_vehicle_action":
            if self.catalog is None:
                from .catalog import TOOLS
                self.catalog = TOOLS
            spec = self.catalog.get(args["name"])
            if spec is None:
                raise ValueError("未知模拟车辆函数；请先按分类ID用load_skill查看有效函数。")
            schema = spec.parameters if hasattr(spec, "parameters") else spec["parameters"]

            if errors:
                raise ValueError("模拟动作参数无效：" + errors[0].message)
            suggestions.append(deepcopy(args))
            return {"ok": True, "status": "suggested_only", "executed": False,
                    "message": "建议已记录，尚未执行；会话层将依据最新车况校验。"}
        if name == "finish_task":
            urls = args.get("source_urls", list(sources)[:20])
            unknown = [url for url in urls if url not in sources]

            # correct structured list must not hide an invented URL in summary.
            inline_urls = [url.rstrip(".,;:，。；：！？!?）】》*`")
                           for url in re.findall(r"https?://[^\s<>\"\]\)）】》」』，。；！？]+",

                raise ValueError("无法引用未经本任务真实工具查询的来源 URL。")

            extracts = [item for item in observations if item["tool"] == "extract_web"]

                raise ValueError("地理编码来源只能说明地点，不能单独支撑天气预报；请引用实际forecast来源，无法核实则如实说明。")
            if any(source.get("kind") not in {"forecast", "geocoding"} for source in used_sources) and not extracts:
                raise ValueError("已找到真实网页来源，请先extract_web读取核实；提取失败也要如实报告。")

            if used_sources and any(source.get("kind") == "forecast" for source in used_sources) and all(

                evidence = "forecast"
            elif used_sources and all(source.get("extracted") or source.get("verified") for source in used_sources):
                evidence = "verified" if any(source.get("kind") == "forecast" for source in used_sources) else "extracted"
            if used_sources and evidence == "none":
                evidence = "search_only"
                summary += "\n限制：部分来源仅有搜索摘要，页面未成功提取，相关信息尚未核实。"
            elif not used_sources and web_attempts:

            # with no sentence/character cutoff or second rewrite model call.
            # Original reports and raw model/tool output remain in result/trace.
            spoken_summary = brief_result(args.get("spoken_summary") or args["summary"])

                spoken_summary = brief_result(args["summary"])

            elif not used_sources and web_attempts and not re.search(

                spoken_summary += "这些是动作建议，尚未执行。"
            queried = [source.get("queried_at") for source in used_sources if source.get("queried_at")]
            return {"ok": True, "result": {"summary": summary, "spoken_summary": spoken_summary,
                    "sources": used_sources,
                    "suggested_actions": deepcopy(suggestions), "query_times": sorted(set(queried)),
                    "web_evidence": evidence,
                    "forecast_records": [deepcopy(item["record"]) for item in observations
                        if item["tool"] == "get_weather_forecast" and item.get("ok") and item["record"]["source_url"] in urls],
                    "tool_observations": deepcopy(observations)}}
        raise ValueError("Unknown tool")

def validate_vehicle_plan(plan, request_text, vehicle, allowed_tools=None, completed_operations=None, history=None):
    """Return a validated private plan or raise ValueError; never execute live.

    Source-span coverage is only a text-accounting check. The model can still
    misunderstand a correctly quoted span; external intent gold remains needed.
    """
    if not isinstance(plan, dict) or not isinstance(request_text, str) or not request_text.strip():
        raise ValueError('invalid_plan_or_request')
    if set(plan) - set(PLAN_FIELDS) - PLAN_METADATA_FIELDS:
        raise ValueError('unexpected_plan_field')
    if not isinstance(vehicle, dict) or not _finite(vehicle) or not _finite(plan):
        raise ValueError('nonfinite_or_invalid_state')
    if allowed_tools is None:
        skills = plan.get('loaded_skills', [])
        if not isinstance(skills, list) or any(skill not in SKILLS for skill in skills):
            raise ValueError('invalid_loaded_skills')
        allowed = _names(skills)
    else:
        allowed = list(allowed_tools)
        if not allowed or any(name not in TOOLS for name in allowed):
            raise ValueError('invalid_allowed_tools')
    submitted = {key: deepcopy(plan.get(key)) for key in PLAN_FIELDS}
    for intent in submitted.get('intents') or []:
        if isinstance(intent, dict):
            intent.setdefault('completed_operation_ids', [])
            alias = intent.pop('text', intent.get('source_text'))
            if alias != intent.get('source_text'):
                raise ValueError('source_text_alias_mismatch')
    try:
        Draft202012Validator(plan_schema(allowed, request_text, vehicle)).validate(submitted)
    except ValidationError as exc:
        raise ValueError(f'plan_schema: {exc.message}') from None
    intents, actions = submitted['intents'], submitted['actions']
    by_intent = {intent['id']: intent for intent in intents}
    if len(by_intent) != len(intents):
        raise ValueError('duplicate_intent_id')
    spans, covered = set(), set()
    for intent in intents:
        start, end = intent['source_start'], intent['source_end']
        if not start < end or request_text[start:end] != intent['source_text'] or not intent['source_text'].strip():
            raise ValueError('intent_source_not_exact')
        if (start, end) in spans:
            raise ValueError('duplicate_intent_source_span')
        spans.add((start, end)); covered.update(range(start, end))
    needed = {i for i, character in enumerate(request_text)
              if not character.isspace() and unicodedata.category(character)[0] not in {'P', 'Z'}}
    if not needed <= covered:
        raise ValueError('unaccounted_request_text')
    ordered = _ordered(actions)
    all_action_ids = {action['id'] for action in actions}
    for action in actions:
        if action['intent_id'] not in by_intent or by_intent[action['intent_id']]['status'] != 'planned':
            raise ValueError('action_has_no_planned_intent')
    completed = _completed(completed_operations)
    used_completed, seen_actions = set(), set()
    for action in actions:
        signature = _canonical([action['name'], action['arguments']])
        if signature in seen_actions:
            raise ValueError('duplicate_action')
        if any(signature == _canonical([operation['name'], operation['arguments']]) for operation in completed.values()):
            raise ValueError('already_completed_action')
        seen_actions.add(signature)
    declared_keep = set()
    for intent in intents:
        bound = {action['id'] for action in actions if action['intent_id'] == intent['id']}
        if set(intent['action_ids']) != bound or not set(intent['action_ids']) <= all_action_ids:
            raise ValueError('intent_action_binding_mismatch')
        status = intent['status']
        if status in {'planned', 'keep', 'completed'} and intent['question'].strip():
            raise ValueError('resolved_intent_has_unanswered_question')
        if status == 'planned' and (not bound or intent['keep_paths'] or intent['completed_operation_ids']):
            raise ValueError('planned_intent_missing_action_or_mixed_status')
        if status == 'keep' and (bound or not intent['keep_paths'] or intent['completed_operation_ids']):
            raise ValueError('keep_intent_missing_path_or_mixed_status')
        if status == 'completed':
            ids = intent['completed_operation_ids']
            if bound or intent['keep_paths'] or not ids or any(identity not in completed for identity in ids):
                raise ValueError('completed_intent_has_no_real_receipt')
            if set(ids) & used_completed:
                raise ValueError('completed_operation_reused')
            for identity in ids:
                operation = completed[identity]
                if validate_literal_feature_target(intent['source_text'], operation['name'], operation['arguments'], vehicle):
                    raise ValueError('completed_receipt_target_mismatch')
            used_completed.update(ids)
        elif intent['completed_operation_ids']:
            raise ValueError('completed_receipt_bound_to_wrong_status')
        if status == 'clarification' and (not intent['question'].strip() or bound or intent['keep_paths']):
            raise ValueError('clarification_intent_not_explicit')
        if status == 'refused' and (not intent['reason'].strip() or bound or intent['keep_paths']):
            raise ValueError('refused_intent_not_explicit')
        declared_keep.update(intent['keep_paths'])
    if set(submitted['keep_paths']) != declared_keep:
        raise ValueError('keep_paths_not_bound_to_intents')
    if submitted['status'] == 'planned':
        if any(intent['status'] not in {'planned', 'keep', 'completed'} for intent in intents):
            raise ValueError('planned_with_unresolved_intent')
        if submitted['question'].strip():
            raise ValueError('planned_has_unanswered_question')
    elif actions or any(intent['status'] == 'planned' for intent in intents):
        raise ValueError('nonplanned_cannot_execute_partial_actions')
    elif submitted['status'] == 'clarification' and (not submitted['question'].strip() or not any(i['status'] == 'clarification' for i in intents)):
        raise ValueError('missing_clarification_question')
    elif submitted['status'] == 'refused' and (not submitted['reason'].strip() or not any(i['status'] == 'refused' for i in intents)):
        raise ValueError('missing_refusal_reason')
    candidate = deepcopy(vehicle)
    keep_before = {path: deepcopy(_value(vehicle, path)) for path in declared_keep}
    simulations = []
    for action in ordered:
        name, arguments = action['name'], action['arguments']
        validate_vehicle_action_keep(name, arguments, candidate, declared_keep)
        guard = (validate_required_origin(request_text, name, arguments, _user_history(history, request_text))
                 or validate_literal_feature_target(by_intent[action['intent_id']]['source_text'], name, arguments, candidate)
                 or validate_literal_feature_target(request_text, name, arguments, candidate)
                 or validate_literal_window_hold(request_text, name, arguments, candidate)
                 or validate_literal_authorization(request_text, name, arguments, candidate))
        if guard:
            raise ValueError(f'literal_guard:{guard["kind"]}: {guard["reason"]}')
        if name == 'set_navigation_destination':
            _navigation_literal(by_intent[action['intent_id']]['source_text'], arguments['destination'])
        outcome = execute_tool(name, arguments, candidate)
        if not outcome['ok']:
            raise ValueError(f'simulation_refused: {action["id"]}: {outcome.get("reason")}')
        if any(_canonical(_value(candidate, path)) != _canonical(value) for path, value in keep_before.items()):
            raise ValueError('simulation_changed_keep_path')
        simulations.append({'action_id': action['id'], 'intent_id': action['intent_id'], 'name': name,
                            'arguments': deepcopy(arguments), 'ok': True, 'simulated': True, 'executed': False,
                            'value': deepcopy(outcome.get('value'))})
    validated = deepcopy(plan)
    validated.update(submitted, actions=deepcopy(ordered), ok=submitted['status'] == 'planned', executed=False)
    for intent in validated['intents']:
        intent['text'] = intent['source_text']
    validated['simulation'] = {'executed': False, 'simulated': True, 'actions': simulations,
                               'final_state': candidate}
    validated['validation'] = {'submitted_action_order': [action['id'] for action in actions],
                               'execution_order': [action['id'] for action in ordered],
                               'structurally_valid': True, 'source_spans_accounted': True,
                               'semantic_coverage_proven': False}
    validated['action_ledger'] = [{'action_id': action['id'], 'intent_id': action['intent_id'],
                                  'status': 'planned', 'executed': False} for action in ordered]
    return validated

def _resolve_source_spans(submission, request_text):
    """Add internal offsets only for a unique exact (including overlap) match.

    No original source text, action name, argument or constraint is rewritten.
    The unchanged strict internal validator checks the resulting full ledger.
    """
    resolved = deepcopy(submission)
    if not isinstance(resolved, dict) or not isinstance(resolved.get('intents'), list):
        raise ValueError('invalid_source_intent_ledger')
    for intent in resolved['intents']:
        if not isinstance(intent, dict) or not isinstance(intent.get('source_text'), str) or not intent['source_text'].strip():
            raise ValueError('intent_source_not_exact')
        if 'source_start' in intent or 'source_end' in intent:
            raise ValueError('wire_source_offsets_not_allowed')
        source, positions, cursor = intent['source_text'], [], 0
        while True:
            found = request_text.find(source, cursor)
            if found < 0:
                break
            positions.append(found)
            cursor = found + 1
        if not positions:
            raise ValueError('intent_source_not_exact')
        if len(positions) != 1:
            raise ValueError('intent_source_ambiguous: use a longer unique exact original substring')
        intent['source_start'] = positions[0]
        intent['source_end'] = positions[0] + len(source)
    return resolved

"""Narrow literal refusal layers for covered vehicle authorization conflicts.

None means this guard found no covered literal conflict, not that it proved
intent authorization. It never selects a function, modifies its arguments,
executes an action or infers the current vehicle state from user/model text.
"""
from __future__ import annotations

from copy import deepcopy
import re
from typing import Any

_TARGET = r"氛围灯(?:\s*(?:的\s*)?(?:电源|开关))?"
_OPEN = r"(?:打开|开启|开(?!关))"
_CLOSE = r"(?:关闭|关掉|熄灭|关着|关(?!键))"
_HOLD_PATTERNS = (
    rf"(?:先\s*)?(?:不要|别)\s*{_OPEN}\s*{_TARGET}",
    rf"(?:先\s*)?(?:不要|别)\s*把\s*{_TARGET}\s*{_OPEN}",
    rf"{_TARGET}\s*(?:先\s*)?(?:不要|别)\s*{_OPEN}",
    rf"{_TARGET}\s*(?:先\s*)?(?:别动|不要动|保持原状|保持现状|保持不变|维持原状)",
    rf"(?:先\s*)?(?:别动|不要动)\s*{_TARGET}",
    rf"(?:保持|维持)\s*{_TARGET}\s*(?:原状|现状|不变)",
)
_SETTING_PATTERNS = (
    ("explicit_enable", rf"{_OPEN}\s*{_TARGET}"),
    ("explicit_enable", rf"{_TARGET}\s*(?:设为|设置为|改为|保持)?\s*{_OPEN}"),
    ("explicit_disable", rf"{_CLOSE}\s*{_TARGET}"),
    ("explicit_disable", rf"{_TARGET}\s*(?:设为|设置为|改为|保持)?\s*{_CLOSE}"),
    ("explicit_disable", rf"{_TARGET}\s*(?:不要亮了|别亮了)"),
)
# A local correction/negation immediately before a quoted action is not an
# affirmative instruction. This handles “不是关闭X，是打开X” without routing.
_NEGATED_PREFIX = re.compile(r"(?:不是(?:说|要|说要)?|并非|并不是(?:要)?|不要|别|不想|不需要|无需)\s*(?:把\s*)?$")


def _negated(text: str, start: int) -> bool:
    return bool(_NEGATED_PREFIX.search(text[max(0, start - 16):start]))


def _evidence(text: str, patterns, meaning: str) -> list[dict[str, Any]]:
    values = []
    for pattern in patterns:
        for match in re.finditer(pattern, text):
            if not _negated(text, match.start()):
                values.append({'start': match.start(), 'end': match.end(),
                               'text': match.group(), 'meaning': meaning})
    return values


def _snapshot_value(snapshot: Any) -> tuple[bool, Any]:
    if not isinstance(snapshot, dict) or not isinstance(snapshot.get('lights'), dict):
        return False, None
    lights = snapshot['lights']
    return ('ambient' in lights, deepcopy(lights.get('ambient')))


def validate_literal_authorization(text: str, tool_name: str, arguments: dict,
                                   snapshot: dict) -> dict[str, Any] | None:
    """Refuse covered power holds/conflicts, with exact source spans as evidence.

    Only ``set_ambient_light`` writes ``lights.ambient`` here. Color and
    brightness changes have separate properties and never match this guard.
    An explicit hold refuses even a same-value power write: “don't open” is
    not permission to send ``enabled=false``. Contradictory affirmative power
    settings or a literal setting that opposes the supplied boolean need
    clarification; no replacement boolean/tool is returned.
    """
    if tool_name != 'set_ambient_light' or not isinstance(text, str) or not text:
        return None
    if not isinstance(arguments, dict) or type(arguments.get('enabled')) is not bool:
        return None  # The existing JSON Schema layer owns malformed arguments.
    holds = _evidence(text, _HOLD_PATTERNS, 'keep_current_power')
    settings = []
    for meaning, pattern in _SETTING_PATTERNS:
        for item in _evidence(text, (pattern,), meaning):
            # “打开” nested inside “不要打开” is not a second setting.
            if not any(item['start'] < hold['end'] and hold['start'] < item['end'] for hold in holds):
                settings.append(item)
    # Deduplicate overlaps from equivalent literal forms without changing text.
    evidence = sorted({(item['start'], item['end'], item['meaning']): item
                       for item in holds + settings}.values(), key=lambda item: (item['start'], item['end']))
    targets = {item['meaning'] == 'explicit_enable' for item in settings}
    contradictory = bool(holds and settings) or len(targets) > 1
    mismatch = len(targets) == 1 and arguments['enabled'] not in targets
    if not holds and not contradictory and not mismatch:
        return None
    conflict = contradictory or mismatch
    known, current = _snapshot_value(snapshot)
    reason = ('原句中的氛围灯电源要求与本次设置冲突，需澄清；未执行电源变更。'
              if conflict else '原句要求氛围灯电源保持原状，不构成开启或关闭授权；未执行电源变更。')
    return {'kind': 'literal_authorization_conflict' if conflict else 'literal_keep_state',
            'executed': False, 'needs_clarification': conflict,
            'state_path': 'lights.ambient', 'current_value_known': known, 'current_value': current,
            'requested_value': arguments['enabled'], 'evidence': evidence,
            'reason': reason,
            'spoken_reason': '氛围灯电源要求有冲突，先保持当前状态，请确认是保持不动还是明确开关？'
                if conflict else '氛围灯电源保持原状，没有调整。'}


# These explicitly covered feature families are rejection boundaries, not a
# tool router. Lighting comparisons do not broaden the original comfort group.
# Keeping the mapping here independent of catalog dispatch means no substitute
# operation or arguments can be produced by this module.
_FEATURES = {
    'set_mirror_heating': ('mirror_heating', '后视镜加热', r'后视镜\s*(?:的\s*)?加热'),
    'set_steering_heating': ('steering_heating', '方向盘加热', r'方向盘\s*(?:的\s*)?加热'),
    'set_seat_heating': ('seat_heating', '座椅加热', r'座椅\s*(?:的\s*)?加热'),
    'set_seat_ventilation': ('seat_ventilation', '座椅通风', r'座椅\s*(?:的\s*)?通风'),
    'set_seat_massage': ('seat_massage', '座椅按摩', r'座椅\s*(?:的\s*)?按摩'),
    'set_ambient_light': ('ambient_power', '氛围灯电源', _TARGET),
    'set_interior_light': ('interior_light', '车内照明灯', r'车内(?:照明)?灯'),
}
_FEATURE_FAMILIES = (
    frozenset({'mirror_heating', 'steering_heating', 'seat_heating', 'seat_ventilation', 'seat_massage'}),
    frozenset({'ambient_power', 'interior_light'}),
)
_FEATURE_LITERAL = re.compile('|'.join(
    rf'(?P<{feature}>{pattern})' for feature, _, pattern in _FEATURES.values()))
_FEATURE_CLAUSE = re.compile(r'[^，,。；;！？!?\n]+')
_FEATURE_DIRECTIVE = re.compile(
    r'打开|开启|关闭|关掉|关上|启用|停用|启动|停止|取消|设置|设为|设成|改为|'
    r'调到|调至|调成|调高|调低|增加|降低|减少|保持')
_FEATURE_NEGATION = re.compile(r'不是(?:说|要|说要)?|并非|并不是(?:要)?|不要|别|不想|不需要|无需')
_FEATURE_CONTRAST = re.compile(r'而是|但是|但|不过')
_FEATURE_QUESTION = re.compile(r'吗|么|有没有|是否|是不是|还是')
_FEATURE_CORRECTION = re.compile(r'^\s*(?:而是|是|应该是|我要的是)')
_FEATURE_RETRACT = re.compile(r'^\s*(?:不|不对|说错了|我说错了|更正一下)\s*$')
_LIGHT_POWER_DIRECTIVE = re.compile(r'打开|开启|关闭|关掉|关上|启用|停用|保持')


def _requested_feature_evidence(text: str, feature_names: frozenset[str],
                                directive_pattern: re.Pattern) -> list[dict[str, Any]]:
    """Read exact affirmative feature spans without interpreting parameters.

    Only explicit directive clauses or literal feature-only corrections count.
    Locally negated alternatives and state questions do not grant a write.
    The small correction handling is deliberately bounded; an uncovered phrase
    produces no authorization proof and remains the model's responsibility.
    """
    evidence: list[dict[str, Any]] = []
    earlier_negated = False
    for clause_match in _FEATURE_CLAUSE.finditer(text):
        clause = clause_match.group()
        if _FEATURE_RETRACT.fullmatch(clause):
            # “关闭X，不，关闭Y” explicitly retracts the preceding target.
            evidence.clear()
            earlier_negated = True
            continue
        matches = [match for match in _FEATURE_LITERAL.finditer(clause) if match.lastgroup in feature_names]
        if not matches or _FEATURE_QUESTION.search(clause):
            continue
        directive = bool(directive_pattern.search(clause))
        corrective = earlier_negated and bool(_FEATURE_CORRECTION.match(clause))
        feature_only = len(matches) == 1 and clause.strip() == matches[0].group()
        negative_scope = False
        previous_end = 0
        for index, match in enumerate(matches):
            prefix = clause[previous_end:match.start()]
            negatives = list(_FEATURE_NEGATION.finditer(prefix))
            contrasts = list(_FEATURE_CONTRAST.finditer(prefix))
            if contrasts and (not negatives or contrasts[-1].start() > negatives[-1].start()):
                negative_scope = False
            elif negatives:
                negative_scope = True
            next_start = matches[index + 1].start() if index + 1 < len(matches) else len(clause)
            suffix = clause[match.end():next_start]
            suffix_negative = _FEATURE_NEGATION.search(suffix)
            suffix_negated = bool(suffix_negative and re.fullmatch(
                r'\s*(?:先\s*)?', suffix[:suffix_negative.start()]))
            if negative_scope or suffix_negated:
                earlier_negated = True
            elif directive or corrective or feature_only or (earlier_negated and contrasts):
                start, end = clause_match.start() + match.start(), clause_match.start() + match.end()
                evidence.append({'start': start, 'end': end, 'text': text[start:end],
                                 'meaning': 'requested_feature', 'feature': match.lastgroup})
            previous_end = match.end()
    return evidence


def validate_literal_feature_target(text: str, tool_name: str, arguments: dict,
                                    snapshot: dict) -> dict[str, Any] | None:
    """Refuse a covered write aimed at another explicitly requested feature.

    This does not choose the matching operation, authorize its parameters or
    infer intent for pronouns. Multiple explicitly requested features are all
    permitted by this target-only check. The supplied arguments and snapshot
    are never read as user intent or modified.
    """
    selected = _FEATURES.get(tool_name)
    if selected is None or not isinstance(text, str) or not text:
        return None
    family = next(features for features in _FEATURE_FAMILIES if selected[0] in features)
    directive_pattern = _LIGHT_POWER_DIRECTIVE if selected[0] in _FEATURE_FAMILIES[1] else _FEATURE_DIRECTIVE
    evidence = _requested_feature_evidence(text, family, directive_pattern)
    requested = {item['feature'] for item in evidence}
    if not requested or selected[0] in requested:
        return None
    labels = {feature: label for feature, label, _ in _FEATURES.values()}
    requested_labels = list(dict.fromkeys(labels[item['feature']] for item in evidence))
    reason = (f'原句明确请求“{"、".join(requested_labels)}”，本次写操作指向“{selected[1]}”；'
              '部位或功能不一致，未执行。请重新核对原句中的目标。')
    return {'kind': 'literal_target_mismatch', 'executed': False, 'retry_model': True,
            'needs_clarification': False, 'selected_feature': selected[0],
            'evidence': evidence, 'reason': reason,
            'spoken_reason': '部位或功能不一致，这个操作没有执行。'}


_WINDOW_ALIASES = {
    'driver': r'(?:主驾驶(?:位|座|席)?|主驾(?:位|座)?|驾驶位|驾驶席|司机侧|左前(?:排)?|前左|left_front|driver)',
    'front_passenger': r'(?:副驾驶(?:位|座|席)?|副驾(?:位|座)?|右前(?:排)?|前右|right_front|front_passenger)',
    'rear_left': r'(?:后排左侧|后排左边|左后(?:排)?|后左|rear_left)',
    'rear_right': r'(?:后排右侧|后排右边|右后(?:排)?|后右|rear_right)',
}
_WINDOW_CONTEXT = re.compile(r'车窗|窗户|(?<!天)窗')
_WINDOW_SUFFIX = r'\s*(?:的\s*)?(?:车窗|窗户|窗)?'
_WINDOW_KEEP = r'(?:别动|不要动|保持不变|保持原状|保持现状|维持原状|保留原(?:来)?开度)'
_WINDOW_LITERAL = re.compile('|'.join(
    rf'(?P<{window}>{alias})' for window, alias in _WINDOW_ALIASES.items()))
_WINDOW_REFERENT = rf'(?:{"|".join(_WINDOW_ALIASES.values())}){_WINDOW_SUFFIX}'
_WINDOW_GROUP = rf'{_WINDOW_REFERENT}(?:\s*(?:和|及|、|以及)\s*{_WINDOW_REFERENT})*'
_WINDOW_HOLD_PATTERNS = (
    rf'{_WINDOW_GROUP}\s*(?:先\s*)?{_WINDOW_KEEP}',
    rf'{_WINDOW_GROUP}\s*(?:的\s*)?开度\s*(?:保持不变|保持原状|不变|保留)',
    rf'(?:先\s*)?(?:别动|不要动)\s*{_WINDOW_GROUP}',
    rf'(?:保持|维持)\s*{_WINDOW_GROUP}\s*(?:不变|原状|现状)',
    rf'保留\s*{_WINDOW_GROUP}\s*(?:的\s*)?(?:原来|原|当前|现在)?\s*(?:的\s*)?开度',
)


def _window_hold_evidence(text: str) -> list[dict[str, Any]]:
    evidence: list[dict[str, Any]] = []
    for clause_match in _FEATURE_CLAUSE.finditer(text):
        clause = clause_match.group()
        if _FEATURE_RETRACT.fullmatch(clause):
            evidence.clear()
            continue
        for pattern in _WINDOW_HOLD_PATTERNS:
            for match in re.finditer(pattern, clause):
                start, end = clause_match.start() + match.start(), clause_match.start() + match.end()
                if not _negated(text, start):
                    for window_match in _WINDOW_LITERAL.finditer(match.group()):
                        evidence.append({'start': start, 'end': end, 'text': text[start:end],
                                         'meaning': 'keep_current_window_position', 'window': window_match.lastgroup})
    return sorted({(item['start'], item['end'], item['window']): item
                   for item in evidence}.values(), key=lambda item: (item['start'], item['end']))


def validate_literal_window_hold(text: str, tool_name: str, arguments: dict,
                                  snapshot: dict) -> dict[str, Any] | None:
    """Reject a window-position write intersecting an explicit preserved set.

    Seat/side aliases require a literal window context; no thermal preference
    or generic seat hold is inferred. A same-value write is still a write.
    The four existing position writers share this property boundary, while
    child locks, anti-pinch, sunroof and other vehicle properties are untouched.
    """
    if not isinstance(text, str) or not _WINDOW_CONTEXT.search(text):
        return None
    if tool_name in {'close_all_windows', 'open_all_windows'}:
        written = set(_WINDOW_ALIASES)
    elif tool_name in {'set_window_position', 'adjust_window_position'}:
        if not isinstance(arguments, dict) or arguments.get('window') not in _WINDOW_ALIASES:
            return None  # Invalid arguments belong to the JSON Schema layer.
        written = {arguments['window']}
    else:
        return None
    evidence = _window_hold_evidence(text)
    preserved = [window for window in _WINDOW_ALIASES if any(item['window'] == window for item in evidence)]
    affected = [window for window in preserved if window in written]
    if not affected:
        return None
    windows = snapshot.get('windows') if isinstance(snapshot, dict) else None
    positions = windows.get('positions') if isinstance(windows, dict) else None
    positions = positions if isinstance(positions, dict) else {}
    current = {window: deepcopy(positions[window]) for window in preserved if window in positions}
    return {'kind': 'literal_window_hold', 'executed': False, 'retry_model': True,
            'needs_clarification': False, 'preserved_windows': preserved,
            'affected_windows': affected, 'current_positions': current,
            'unknown_windows': [window for window in preserved if window not in positions],
            'evidence': evidence,
            'reason': '原句要求保留的车窗在本次位置写入集合中；即使开度相同也未授权该写入，未执行。请重新核对保留约束。',
            'spoken_reason': '这个操作会覆盖要求保持不动的车窗，没有执行。'}

def validate_required_origin(text, tool_name, arguments, recent_user_texts=()):
    """Refuse a literal nearest-airport action with no stated origin.

    This covers the location-free navigation forms above, not arbitrary place
    recognition or route planning. A directly preceding explicit user origin
    is sufficient; mentions in research/pages are not a location source.
    """
    if tool_name not in {'delegate_task', 'set_navigation_destination', 'search_places', 'start_poi_navigation'} or not isinstance(text, str):
        return None
    match = _NEAREST_AIRPORT.fullmatch(text)
    if not match:
        return None
    preceding = list(recent_user_texts)[-1:]
    if any(isinstance(value, str) and _EXPLICIT_ORIGIN.fullmatch(value) for value in preceding):
        return None
    return {'kind': 'missing_required_origin', 'executed': False, 'retry_model': True,
            'needs_clarification': False, 'required_condition': '出发城市或地址',
            'evidence': [{'start': match.start(), 'end': match.end(), 'text': match.group(),
                          'meaning': 'nearest_airport_requires_origin'}],
            'reason': '没有可用GPS或用户提供的出发城市/地址，不能查询或选择最近机场、登记任务或启动导航。工具参数中的猜测地点和网页/POI结果不构成用户提供的出发地。请直接只问用户现在在哪个城市或地址，不承诺已接手导航。',
            'spoken_reason': '你现在在哪个城市或地址？'}

"""A display-syntax guard for real speech, never a canned response generator."""
import json
import re


def spoken_number(value):
    number = str(value)
    negative = number.startswith('-')
    if negative: number = number[1:]
    whole, _, fraction = number.partition('.')
    if not whole.isdigit() or int(whole) > 100: return str(value)
    digits = '零一二三四五六七八九'
    n = int(whole)
    if n == 100: words = '一百'
    elif n < 10: words = digits[n]
    else: words = (digits[n // 10] if n >= 20 else '') + '十' + (digits[n % 10] if n % 10 else '')
    fraction = fraction.rstrip('0')
    if fraction: words += '点' + ''.join(digits[int(d)] for d in fraction)
    return ('负' if negative else '') + words


def _json_words(value):
    if isinstance(value, dict):
        labels = {'summary': '结论', 'answer': '回答', 'spoken_summary': '结论',
                  'reason': '原因', 'error': '问题', 'temperature': '温度', 'status': '状态'}
        return '；'.join(f'{labels.get(str(k), str(k))}是{_json_words(v)}' for k, v in value.items())
    if isinstance(value, list):
        return '、'.join(_json_words(item) for item in value)
    if value is True: return '是'
    if value is False: return '否'
    if value is None: return '未提供'
    return str(value)


def _table_words(lines):
    result, index = [], 0
    while index < len(lines):
        line = lines[index]
        if '|' not in line or index + 1 >= len(lines) or not re.fullmatch(r'[\s|:\-]+', lines[index + 1]):
            result.append(line)
            index += 1
            continue
        headers = [v.strip() for v in line.strip().strip('|').split('|')]
        index += 2
        while index < len(lines) and '|' in lines[index]:
            cells = [v.strip() for v in lines[index].strip().strip('|').split('|')]
            result.append('，'.join(f'{headers[i]}是{cell}' if i < len(headers) else cell
                                   for i, cell in enumerate(cells) if cell) + '。')
            index += 1
    return result


def speech_text(text: str) -> str:
    """Remove markup while preserving statements, negation and conditions.

    Called only on complete eligible text, so split URL/Markdown deltas, numbers
    and units cannot leak through an incremental boundary. No semantic length
    cutoff is applied here; the model/receipt/result contract controls length.
    """
    text = str(text or '').strip()
    text = re.sub(r'^```[^\n]*\n?', '', text)
    text = re.sub(r'```\s*$', '', text)
    try:
        parsed = json.loads(text)
    except (ValueError, TypeError):
        pass
    else:
        if isinstance(parsed, (dict, list)):
            text = _json_words(parsed)
    decoder = json.JSONDecoder()
    offset = 0
    while True:
        match = re.search(r'[\{\[]', text[offset:])
        if not match: break
        start = offset + match.start()
        if text[start] == '[' and text[:start].rstrip()[-1:] not in {':', '：', '='}:
            offset = start + 1
            continue
        try:
            value, length = decoder.raw_decode(text[start:])
        except ValueError:
            offset = start + 1
            continue
        if isinstance(value, (dict, list)):
            words = _json_words(value)
            text = text[:start] + words + text[start + length:]
            offset = start + len(words)
        else:
            offset = start + length
    text = re.sub(r'!\[([^\]]*)\]\([^)]*\)', r'\1', text)
    text = re.sub(r'\[([^\]]+)\]\(https?://[^\s)]+\)', r'\1', text)
    had_url = bool(re.search(r'https?://|www\.', text))
    text = re.sub(r'(?:https?://|www\.)[^\s<>，。！？；）)\]]+', '', text)
    text = re.sub(r'[^]*|【[^】]*†[^】]*】|\[(?:\d+(?:\s*[,，-]\s*\d+)*|cite:[^\]]*|turn\d+[^\]]*)\]', '', text)
    text = re.sub(r'(?m)^\s*\[\d+\]:.*$', '', text)
    lines = _table_words(text.splitlines())
    output = []
    for line in lines:
        line = re.sub(r'^\s*(?:#{1,6}\s*|>\s*|[-*+•]\s+|\d+(?:[.)]\s+|、\s*))', '', line)
        line = re.sub(r'[`*_~]+', '', line)
        line = re.sub(r'<[^>]+>', '', line)
        line = re.sub(r'\[([^\]]+)\]', r'\1', line)
        line = line.replace('|', '，')
        line = re.sub(r'\s+', ' ', line).strip(' |\t')
        if not line or re.fullmatch(r'[-:|\s]+', line): continue
        if line.strip('：:。') in {'结论', '调研结果', '查询结果', '详细报告', '来源', '参考来源', '回答', '建议', '核查结果', '结果'}:
            continue
        if line.endswith(('：', ':')):
            line = line[:-1]
        if line:
            output.append(line if line[-1] in '。！？!?；;' else line + '。')
    text = ''.join(output)
    text = re.sub(r'(-?\d+(?:\.\d+)?)\s*(?:℃|°C)', r'\1度', text)
    text = re.sub(r'(-?\d+(?:\.\d+)?)\s*%', r'百分之\1', text)
    text = re.sub(r'[。；]{2,}', '。', text).strip(' 。；')
    text = re.sub(r'^(?:(?:好的|好呀|收到)[，。！\s]*|(?:以下是结果|以下为你整理|查询结果)[：:。\s]*)+', '', text)
    if text: return text + ('' if text[-1] in '。！？!?' else '。')
    return '相关链接在详情里。' if had_url else ''


def brief_result(text: str, max_chars: int = 120, max_sentences: int = 3) -> str:
    """Compatibility guard; length budgets belong to the generation contract.

    A deterministic guard cannot know which later facts qualify an earlier
    recommendation. Preserve all statements when a legacy report lacks its
    own generated spoken summary. The old budget arguments remain accepted
    for callers, but never cause sentence or character truncation.
    """
    return speech_text(text)

"""Lossless archive, compact model context: helpers only transform deep copies.

These functions never select an intent, truncate text, rewrite user constraints,
change a tool argument or modify the audit/DB record. The current vehicle
snapshot and offered tool schemas already supply state and capability details.
"""
from __future__ import annotations

from copy import deepcopy
import json
from typing import Any

TASK_FIELDS = ('id', 'goal', 'status', 'revision', 'summary', 'question', 'error', 'constraints')


def _result_for_context(result: Any, read_details: bool) -> Any:
    if read_details is True or not isinstance(result, dict):
        return deepcopy(result)
    if isinstance(result.get('spoken_summary'), str) and result['spoken_summary'].strip():
        return {'spoken_summary': result['spoken_summary']}
    # Legacy tasks without a generated spoken summary retain their complete
    # answer. Omitting them would discard facts rather than remove duplication.
    if 'summary' in result:
        return {'summary': deepcopy(result['summary'])}
    return deepcopy(result)


def _task_for_context(task: Any, read_details: bool) -> Any:
    if not isinstance(task, dict):
        return deepcopy(task)
    compact = {key: deepcopy(task[key]) for key in TASK_FIELDS if key in task}
    if 'result' in task:
        compact['result'] = _result_for_context(task['result'], read_details)
    return compact


def compact_tool_result(body: Any, tool_name: str | None = None,
                        read_details: bool = False) -> Any:
    """Return an independent prompt copy of one structured tool result.

    Task conditions and status remain complete. Explicit detailed reads retain
    the complete result, while ordinary task references carry the generated
    spoken answer. Skill schemas are omitted only for a recognizable loaded
    manifest: the actual schemas are already offered through ``tools_for``.
    Ordinary tool values, reasons and nested payloads are never shortened.
    """
    result = deepcopy(body)
    if not isinstance(result, dict):
        return result
    result.pop('state', None)
    task = result.get('task')
    if isinstance(task, dict):
        result['task'] = _task_for_context(task, read_details)
    if isinstance(result.get('tasks'), list):
        result['tasks'] = [_task_for_context(task, read_details) for task in result['tasks']]
    if isinstance(result.get('items'), list):
        result['items'] = [compact_tool_result(item, tool_name, read_details) for item in result['items']]
    skill = result.get('skill')
    manifest = (isinstance(skill, dict) and isinstance(skill.get('id'), str)
                and isinstance(skill.get('tool_names'), list)
                and all(isinstance(name, str) for name in skill['tool_names']))
    # Recognizing a catalog manifest also supports old tool messages that
    # don't carry their function name in the result body itself.
    if manifest and (tool_name in (None, 'load_skill')) and isinstance(result.get('tools'), list):
        result.pop('tools')
    return result


def compact_message(record: Any, tool_name: str | None = None,
                    read_details: bool = False) -> Any:
    """Compact valid JSON tool content only; every other message stays exact."""
    result = deepcopy(record)
    if not isinstance(result, dict) or result.get('role') != 'tool':
        return result
    content = result.get('content')
    if not isinstance(content, str):
        return result
    try:
        body = json.loads(content)
    except (ValueError, TypeError):
        return result
    result['content'] = json.dumps(compact_tool_result(body, tool_name, read_details),
                                   ensure_ascii=False, separators=(',', ':'))
    return result
